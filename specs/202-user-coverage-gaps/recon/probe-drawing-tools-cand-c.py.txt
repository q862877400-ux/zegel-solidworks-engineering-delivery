class _ProbeBase(BaseModel):

    model_config = ConfigDict(str_strip_whitespace=True, extra="forbid")

_TOOL_LOCK = threading.Lock()


def _run_locked(operation, response_format, load_automation: bool = True):
    """★ `mcp-server/server.py:1031 _run_locked` 的**同形状**本地替身。

    ★ 只保留工具层探针必需的契约三条：
      ① 跑 `op()`；② **把异常变成 `status="error"` 的 JSON 串**（★ 这正是 server 的行为
         ⇒ 工具函数**不会**把异常抛回调用方 —— ★ 于是「能力层抛了」与「工具层炸了」
         可以靠**响应形状**分开）；③ JSON 序列化。
    ⚠ server 那版还做 `_sw_lock` / `_load_automation_modules()` / `_coinitialize()` /
      按 `response_format` 切 markdown —— ★ 那是**传输层**，与「工具层能否跑通」无关。
    ⚠ ★★ **本探针不 import `server.py`** —— 那会在 import 期就 `register_capabilities(...)`
      整个工具面（`server.py:2468`），★ 副作用超出探针范围，也违反「不启 MCP 单源」。
    """
    with _TOOL_LOCK:
        try:
            return json.dumps(operation(), ensure_ascii=False, indent=2, default=str)
        except Exception as exc:                                   # noqa: BLE001
            return json.dumps({"status": "error", "error_type": type(exc).__name__,
                               "message": str(exc),
                               "suggestion": "确认 SolidWorks 在跑、活动文档是目标工程图、"
                                             "路径是绝对 Windows 路径。"},
                              ensure_ascii=False, indent=2, default=str)


def _collect_expose_specs():
    """★ 取全部 spec（★ 契约要求的入口是 `register_capabilities.all_specs()`），
    只留 `module == "sw_drawing_expose"` 那一批；★ 失败才退回直接 import `tool_specs_batch75`
    —— ★ **并把失败原因记进证据**（★ 不静默换路）。"""
    info = {"expected_module": EXPOSE_MODULE, "used_entry": None, "all_specs_error": None}
    specs = []
    try:
        specs = [s for s in RC.all_specs() if getattr(s, "module", "") == EXPOSE_MODULE]
        info["used_entry"] = "register_capabilities.all_specs()"
    except Exception as exc:                                       # noqa: BLE001
        info["all_specs_error"] = "%s: %s" % (type(exc).__name__, str(exc)[:240])
        import tool_specs_batch75 as B75
        specs = [s for s in B75.SPECS_BATCH75 if getattr(s, "module", "") == EXPOSE_MODULE]
        info["used_entry"] = "tool_specs_batch75.SPECS_BATCH75（all_specs 失败后的回退，已记录原因）"
    info["spec_count"] = len(specs)
    info["spec_names"] = [s.name for s in specs]
    info["missing_expected_tools"] = [n for n in EXPECTED_TOOLS if n not in info["spec_names"]]
    #: ★ 把**契约**（inject / params / 必填性）一并记进证据 —— ★ 探针按它备实参，不靠记忆
    info["contracts"] = {}
    for s in specs:
        info["contracts"][s.name] = {
            "inject": REG.effective_inject(s),
            "module": s.module,
            "function": s.function,
            "params": [{"name": p.name,
                        "kind": getattr(p.kind, "__name__", str(p.kind)),
                        "required": bool(p.is_required)}
                       for p in s.params],
        }
    return specs, info


def _tool_callers(specs) -> dict:
    """★ 由 ToolSpec 造出**可调的工具函数**（★ 这就是 `register()` 内部那条路，只是不进 MCP）。

    ★ 走的是 `tool_registry` 的两个公开入口 `build_input_model` / `make_tool_function`
      —— ★ 与 `register()`（`tool_registry.py:558-559`）**逐字同一条**。
    """
    callers = {}
    for spec in specs:
        model_cls = REG.build_input_model(spec, _ProbeBase, _ResponseFormat)
        fn = REG.make_tool_function(spec, model_cls, _run_locked, resolvers=RC.RESOLVERS)

        def make(fn=fn, model_cls=model_cls, name=spec.name):
            def call(**kwargs):
                return fn(model_cls(response_format="json", **kwargs))
            call.__name__ = name
            return call

        callers[spec.name] = make()
    return callers


def _invoke(call, kwargs) -> dict:
    """★ 调一次**工具层**函数 ⇒ 把「响应形状 / 解析结果 / 调用期异常」**分开**报。"""
    out = {"call_error": None, "response_is_str": False, "payload": None,
           "response_head": None, "json_error": None}
    if call is None:
        out["call_error"] = "★ 该工具不在取到的 spec 里（本批没注册到 all_specs）"
        return out
    try:
        raw = call(**kwargs)
    except Exception as exc:                                       # noqa: BLE001
        out["call_error"] = "%s: %s" % (type(exc).__name__, str(exc)[:240])
        return out
    out["response_is_str"] = isinstance(raw, str)
    text = raw if isinstance(raw, str) else None
    out["response_head"] = (text or (repr(raw)[:600] if raw is not None else None))
    if text is not None:
        try:
            out["payload"] = json.loads(text)
        except Exception as exc:                                   # noqa: BLE001
            out["json_error"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    return out

def _view_name(view) -> str:
    for attr in ("GetName2", "Name"):
        try:
            v = _get(view, attr)
            if v:
                return str(v)
        except Exception:                                          # noqa: BLE001
            continue
    return ""


def _view_chain(doc):
    """★ 视图链 `GetFirstView()` → `GetNextView()`（★ 含「图纸本身」那个伪视图）。"""
    out = []
    try:
        view = _get(doc, "GetFirstView")
    except Exception:                                              # noqa: BLE001
        return out
    n = 0
    while view is not None and n < 500:                            #: ★ 上限防死循环
        n += 1
        out.append(view)
        try:
            view = _get(view, "GetNextView")
        except Exception:                                          # noqa: BLE001
            break
    return out


def _real_views(doc, names):
    """★ **真视图**（★ 排除「图纸本身」那个伪视图 —— ★ 由 `sw_drawing_query.list_views` 判，
    ★ 本探针不自造那个陷阱的判据）。"""
    keep = {str(n or "") for n in names}
    return [v for v in _view_chain(doc) if _view_name(v) in keep]

def _readout(sw, doc, part_path: str = "") -> dict:
    """★ 一份**可回读**的基线 / 终态读数（★ 判据只允许用这里有的量）。"""
    row = {}
    try:
        views = DQ.list_views(doc) or []
    except Exception as exc:                                       # noqa: BLE001
        views = []
        row["views_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    row["view_count"] = len(views)
    row["view_names"] = [str((v or {}).get("name") or "") for v in views]
    chain = _view_chain(doc)
    real = _real_views(doc, row["view_names"])
    counters = {}
    #: ★ 两份都算：「真视图」与「整条链」—— ★ 有些注解挂在**图纸**上，★ 只数真视图会漏
    for label, group in (("real", real), ("all", chain)):
        totals = {}
        for key, member in _COUNTERS:
            vals = []
            for v in group:
                try:
                    vals.append(int(_get(v, member) or 0))
                except Exception:                                  # noqa: BLE001
                    pass
            totals[key] = sum(vals) if vals else None
        totals["view_objects"] = len(group)
        counters[label] = totals
    row["counters"] = counters
    try:
        row["sheet_info"] = DQ.current_sheet_info(doc)
    except Exception as exc:                                       # noqa: BLE001
        row["sheet_info"] = None
        row["sheet_info_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    try:
        row["drawing_path"] = str(_get(doc, "GetPathName") or "")
    except Exception as exc:                                       # noqa: BLE001
        row["drawing_path"] = ""
        row["drawing_path_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:120])
    try:
        row["doc_title"] = str(_get(doc, "GetTitle") or "")
    except Exception:                                              # noqa: BLE001
        row["doc_title"] = ""
    row["selection_counts"] = _sel_counts(doc)
    row["part_saved_with_extension"] = bool(
        part_path and str(part_path).lower().endswith((".sldprt", ".sldasm"))
        and Path(part_path).is_file())
    #: ★ 注入解析器 `resolve_model` 取的是 `ISldWorks.ActiveDoc` —— ★ 这里**回读同一个东西**来核对
    try:
        active = _get(sw, "ActiveDoc")
        row["active_doc_title"] = str(_get(active, "GetTitle") or "") if active is not None else ""
    except Exception as exc:                                       # noqa: BLE001
        row["active_doc_title"] = ""
        row["active_doc_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    row["active_doc_is_target"] = bool(row["active_doc_title"]
                                       and row["active_doc_title"] == row["doc_title"])
    return row


#: ★ 增量取值路径（★ 每个读数给「真视图」与「整条链」两条，★ 取**任一份涨了**就算涨，
#:   —— 与 `tools/prove_drawing_batch.py::_delta` 同口径）
_DELTA_PATHS = {
    "views": [("view_count",)],
    "note": [("counters", "real", "note"), ("counters", "all", "note")],
    "table": [("counters", "real", "table"), ("counters", "all", "table")],
    "anno": [("counters", "real", "anno"), ("counters", "all", "anno")],
    "center": [("counters", "real", "center"), ("counters", "all", "center")],
    "selection": [("selection_counts", "0"), ("selection_counts", "1"),
                  ("selection_counts", "2"), ("selection_counts", "4"),
                  ("selection_counts", "8")],
}


def _pick(doc_like, path):
    cur = doc_like
    for p in path:
        if not isinstance(cur, dict) or p not in cur:
            return None
        cur = cur[p]
    return cur


def _deltas(before, after, keys):
    """★ 逐键算增量 ⇒ `(汇总 dict, 明细 dict)`；★ 任一份量不到就**只记 None**（★ 不猜 0）。"""
    summary, detail = {}, {}
    for key in keys:
        per_path, best = {}, None
        for path in _DELTA_PATHS.get(key, []):
            a, b = _pick(before, path), _pick(after, path)
            if isinstance(a, int) and isinstance(b, int):
                d = b - a
                per_path[".".join(path)] = d
                best = d if best is None else max(best, d)
        summary[key] = best
        detail[key] = per_path
    return summary, detail


def _classify(row) -> str:
    """★ **三态**：建成 / 前置未成立 / 真阴性（★ 前置不成立 ⇒ 绝不判「能力不成」）。"""
    if not row.get("preconditions_held"):
        return "precondition_not_held"
    for value in (row.get("deltas") or {}).values():
        if isinstance(value, int) and value >= 1:
            return "built"
    for value in (row.get("signals") or {}).values():
        if value is True:
            return "built"
    return "true_negative"

def _build_part(sw, row):
    """★ 建一块**带孔**的板并**存盘**（★ 存盘是硬要求，见下）。

    ★★ 官方 `CreateDrawViewFromModelView3` 的 Remarks 逐字要求 `ModelName` 是
      **完整路径且含 `.sldprt`/`.sldasm` 扩展名** ⇒ ★ 不存盘就没法测投影视图。
    """
    doc = C.new_document(sw, "part")
    try:
        P.start_sketch(doc, plane_name="Front Plane")
        _get(_get(doc, "SketchManager"), "CreateCornerRectangle",
             -0.04, -0.03, 0.0, 0.04, 0.03, 0.0)
        P.end_sketch(doc)
        names = [n for n in (P.feature_names(doc) or [])
                 if n.startswith("草图") or n.lower().startswith("sketch")]
        if not names:
            row["extrude_err"] = "找不到草图特征名（feature_names 没回草图）"
            return None
        P.extrude_boss(doc, names[-1], 0.01)
        row["extrude_sketch"] = names[-1]
    except Exception as exc:                                       # noqa: BLE001
        row["extrude_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:200])
        return None
    try:
        from scripts import sw_hole_features as HF
        rep = HF.create_hole_wizard_at_selection_report(doc, "+z", "M8")
        row["hole"] = {k: rep.get(k) for k in ("ok", "selection_mark") if k in rep}
    except Exception as exc:                                       # noqa: BLE001
        #: ★ 孔不成**不阻断**：只是孔表/中心符号线那两格的前置更弱 ⇒ 如实记下来
        row["hole_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:200])
    try:
        row["saved"] = bool(C.save_document(doc, str(PART_PATH)))
    except Exception as exc:                                       # noqa: BLE001
        row["save_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:200])
        row["saved"] = False
    return doc if row.get("saved") else None


def _activate_view(doc, name) -> bool:
    """★ 激活视图（`IDrawingDoc.ActivateView(string)`）。"""
    try:
        return bool(_get(doc, "ActivateView", str(name)))
    except Exception:                                              # noqa: BLE001
        return False


def _outline(view):
    """★ 视图包围盒（官方 `IView.GetOutline`：**4 个 double，单位米**）。"""
    try:
        vals = [float(x) for x in _get(view, "GetOutline")]
    except Exception:                                              # noqa: BLE001
        return None
    return vals if len(vals) >= 4 else None


def _ensure_parent_view(sw, doc, row, call_projected):
    """★ 依赖视图的三格需要一个**母视图**。

    ① 先**用被测工具本身**建（`solidworks_drawing_create_projected_view`）—— ★ 最贴近真实用法；
    ② 它没建出来，才走**直调** `IDrawingDoc.CreateDrawViewFromModelView3` 兜底，
       ★ 视图名**向模型问**（`IModelDoc2.GetModelViewNames`）—— 官方举例的 `"*Front"` 是
         **英文界面**的取值，★ 中文界面照抄必回 `None`（事实 494）。
    ★ ★ 兜底只是**前置准备**，★ **判据仍然是「被测工具」那一次的读数增量**，★ 兜底与否已逐格标注。
    """
    if call_projected is not None:
        row["setup_projected_view_tool"] = _invoke(
            call_projected, {"part_path": str(PART_PATH), "projection": "first_angle"})
    names = [str((v or {}).get("name") or "") for v in (DQ.list_views(doc) or [])]
    if names:
        row["setup_parent_view"] = names[0]
        row["setup_parent_view_source"] = "被测工具 solidworks_drawing_create_projected_view"
        return names[0]
    view_name = None
    try:
        names_mv = [str(x) for x in (_get(doc, "GetModelViewNames") or [])]
        row["model_view_names"] = names_mv
        view_name = names_mv[0] if names_mv else None
    except Exception as exc:                                       # noqa: BLE001
        row["model_view_names_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    if view_name:
        try:
            _get(doc, "CreateDrawViewFromModelView3", str(PART_PATH), view_name,
                 0.14, 0.12, 0.0)
        except Exception as exc:                                   # noqa: BLE001
            row["setup_fallback_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:200])
    names = [str((v or {}).get("name") or "") for v in (DQ.list_views(doc) or [])]
    if names:
        row["setup_parent_view"] = names[0]
        row["setup_parent_view_source"] = ("★ 直调 CreateDrawViewFromModelView3 的**前置兜底**"
                                           "（★ 非判据；★ 兜底只用官方例子给的调用序）")
        return names[0]
    row["setup_parent_view"] = None
    row["setup_parent_view_source"] = "★ 两条路都没建出母视图"
    return None


def _prep_section_line(sw, doc, row, view_name) -> None:
    """★ 预备剖面线 —— ★ 照官方例子 `Create_Section_View_at_Specified_Location_Example_VB`
    的**调用序**：`ActivateView` → 选视图 → `ClearSelection2` → `CreateLine2`
    （★ 剖面线就是活动视图里画的一条草图线）。
    ★ 画完**回读计数** —— ★ 选不中就是「前置未成立」，★ **不是**「剖面视图不成」。
    """
    row["section_activated"] = _activate_view(doc, view_name)
    views = _real_views(doc, [view_name])
    view = views[0] if views else None
    bbox = _outline(view) if view is not None else None
    row["section_view_outline_m"] = [round(v, 6) for v in bbox] if bbox else None
    if not bbox:
        row["section_prep_note"] = "★ 取不到视图包围盒 ⇒ 画不出剖面线（前置未成立）"
        return
    y_mid = (bbox[1] + bbox[3]) / 2.0
    p1 = (bbox[0] + 0.1 * (bbox[2] - bbox[0]), y_mid)
    p2 = (bbox[2] - 0.1 * (bbox[2] - bbox[0]), y_mid)
    try:
        _get(_get(doc, "SketchManager"), "CreateLine2",
             float(p1[0]), float(p1[1]), 0.0, float(p2[0]), float(p2[1]), 0.0)
        row["section_line_drawn_m"] = [round(p1[0], 6), round(p1[1], 6),
                                       round(p2[0], 6), round(p2[1], 6)]
    except Exception as exc:                                       # noqa: BLE001
        row["section_line_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])


def _prep_detail_circle(sw, doc, row, view_name) -> None:
    """★ 预备**局部圆** —— ★ 官方例子 `Create_Detail_Circle_and_Detail_View_Example_VB` 的
    调用序是 `ActivateView` → `SketchManager.CreateCircle(...)` → 才调 `CreateDetailViewAt4`。"""
    row["detail_activated"] = _activate_view(doc, view_name)
    views = _real_views(doc, [view_name])
    view = views[0] if views else None
    bbox = _outline(view) if view is not None else None
    row["detail_view_outline_m"] = [round(v, 6) for v in bbox] if bbox else None
    if not bbox:
        row["detail_prep_note"] = "★ 取不到视图包围盒 ⇒ 画不出局部圆（前置未成立）"
        return
    cx = (bbox[0] + bbox[2]) / 2.0
    cy = (bbox[1] + bbox[3]) / 2.0
    r = 0.2 * min(bbox[2] - bbox[0], bbox[3] - bbox[1])
    try:
        _get(_get(doc, "SketchManager"), "CreateCircle",
             float(cx), float(cy), 0.0, float(cx + r), float(cy), 0.0)
        row["detail_circle_m"] = {"center": [round(cx, 6), round(cy, 6)], "r_m": round(r, 6)}
    except Exception as exc:                                       # noqa: BLE001
        row["detail_circle_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])


def _find_sheet_format():
    """★ 找一个 `.slddrt` 图纸格式模板（★ 优先官方举例的横向 A 幅 `a - landscape.slddrt`）。"""
    for directory in SHEET_FORMAT_DIRS:
        candidate = Path(directory) / "a - landscape.slddrt"
        if candidate.is_file():
            return str(candidate)
    return None


# ── 一格一格跑 ───────────────────────────────────────────────────────
def _case(sw, row, doc, callers, tool, kwargs, *, precond, deltas=("views",),
          part_path: str = "", call_even_if_pre_fails: bool = False):
    """★ 一格 = 一次**工具层**调用 + 前后两次读数 ⇒ 把三态与耗时写进 `row`。

    ★ ★ **前置不成立时默认不调**（★ 避免制造假阴性）；★ `call_even_if_pre_fails=True`
      的那几格是**故意调**的 —— ★ 目的是验「工具层确实把 `preconditions_held` 透出来」
      （事实 496：前置与能力必须分开记），★ 判读仍按前置未成立处理。
    """
    row["tool"] = tool
    row["params"] = {k: (v if isinstance(v, (str, int, float, bool, list)) else repr(v))
                     for k, v in kwargs.items()}
    row["tool_present"] = tool in callers
    before = _readout(sw, doc, part_path)
    detail, pre_ok = precond(sw, doc, before, row)
    row["preconditions"] = detail
    row["preconditions_held"] = bool(pre_ok)
    row["before"] = before
    if not pre_ok and not call_even_if_pre_fails:
        row["state"] = "precondition_not_held"
        row["ok"] = False
        row["blocked"] = ("★ 前置未成立 ⇒ 本格**不调用**工具，★ 不能当「能力不成」的证据"
                          "（★ 缺什么见 preconditions）")
        return row
    t0 = time.time()
    got = _invoke(callers.get(tool), kwargs)
    row["seconds"] = round(time.time() - t0, 2)
    row["tool_call_error"] = got["call_error"]
    row["tool_response_is_str"] = got["response_is_str"]
    row["tool_response_head"] = got["response_head"]
    row["tool_response_json_error"] = got["json_error"]
    payload = got["payload"] if isinstance(got["payload"], dict) else {}
    row["tool_payload_ok"] = payload.get("ok")
    row["tool_payload_op"] = payload.get("op")
    row["tool_payload_verified_by"] = payload.get("verified_by")
    row["tool_payload_preconditions_held"] = payload.get("preconditions_held")
    is_error = payload.get("status") == "error"
    row["tool_response_is_error"] = bool(is_error)
    row["tool_error"] = ("%s: %s" % (payload.get("error_type"), str(payload.get("message"))[:240])
                         if is_error else None)
    raw = payload.get("raw")
    raw_dict = raw if isinstance(raw, dict) else {}
    row["capability_raw_status"] = raw_dict.get("status")
    row["capability_raw_created"] = raw_dict.get("created")
    row["capability_raw_error_code"] = raw_dict.get("error_code")
    row["capability_raw_is_dict"] = isinstance(raw, dict)
    after = _readout(sw, doc, part_path)
    row["after"] = after
    summary, detail_paths = _deltas(before, after, deltas)
    row["deltas"] = summary
    row["delta_paths"] = detail_paths
    sheet_changed = bool(before.get("sheet_info") != after.get("sheet_info"))
    row["signals"] = {
        "capability_raw_created": raw_dict.get("created") is True,
        "capability_raw_status_pass": raw_dict.get("status") == "pass",
        "capability_raw_verified": raw_dict.get("verified") is True,
        "sheet_info_changed": sheet_changed,
    }
    row["sheet_readback_changed"] = sheet_changed
    row["state"] = _classify(row)
    row["ok"] = row["state"] == "built"
    #: ★★ **工具自报的 `ok` 与独立读数是否一致** —— ★ 不一致**本身就是**要报的东西，
    #:   ★ 本探针**不改**任何一行，只把事实摆出来（★ 事实：能力层有几处回 dict 而非对象，
    #:   ★ 而 `sw_drawing_expose._result` 的 `ok` 推导对「回 dict 的能力」不成立）。
    row["tool_ok_vs_readback"] = {
        "tool_ok": row["tool_payload_ok"],
        "state": row["state"],
        "agree": (bool(row["tool_payload_ok"]) == bool(row["ok"])),
    }
    if row["state"] == "precondition_not_held":
        row["blocked"] = ("★ 前置未成立（★ 本格是**故意**调了一次，★ 为验工具层有没有把 "
                          "`preconditions_held` 透出来）⇒ **不得**据此判「能力不成」")
    return row


def _run_case(sw, callers, *, label, tool, kwargs, setup=None, precond,
              deltas=("views",), part_path: str = "", call_even_if_pre_fails: bool = False):
    """★ 一格的壳：新建工程图 → 可选前置准备 → `_case` → **关掉自己开的**图纸。"""
    row = {"label": label}
    doc = None
    try:
        doc = _new_drawing(sw, row)
        if doc is None:
            row["state"] = "precondition_not_held"
            row["ok"] = False
            row["blocked"] = "★ 前置未成立：新建工程图失败（★ 见 new_drawing_err）"
            return row
        if setup is not None:
            setup(sw, doc, row)
        _case(sw, row, doc, callers, tool, kwargs, precond=precond, deltas=deltas,
              part_path=part_path, call_even_if_pre_fails=call_even_if_pre_fails)
    except Exception as exc:                                       # noqa: BLE001
        row["harness_error"] = "%s: %s" % (type(exc).__name__, str(exc)[:240])
        row["state"] = "harness_error"
        row["ok"] = False
        row["blocked"] = "★ 探针自身出错 ⇒ **不算**能力判据（★ 与三态分开报）"
    finally:
        if doc is not None:
            _close(sw, doc)
    return row

def case_t1_projected(sw, callers):
    """★ T1 投影视图：`solidworks_drawing_create_projected_view`。

    ★ 实参**逐字按 spec.params**：`part_path`(str, 必填) + `projection`(str, 必填)。

def case_t2_section(sw, callers):
    """★ T2 剖面视图：`solidworks_drawing_create_section_view`（`CreateSectionViewAt5`）。

    ★ 实参逐字按 spec.params：`x_mm` + `y_mm`（都是**毫米**、都必填）。
    ★ 前置（★ 官方 Remarks 逐字）：**调用前必须先选好剖面线** —— ★ 剖面线是活动视图里
      画的一条草图线（官方例子的调用序）。★ 本探针照那个序准备，然后**回读选区计数**。
    ★ ★ 前置不成立也**照调一次**（`call_even_if_pre_fails=True`）：★ 目的是验工具层
      **有没有把 `preconditions_held` 透出来**，★ 判读仍按前置未成立处理。
    ★ 判据：**视图数增量 ≥ 1**（★ 前置成立才算）。
    """
    def setup(sw_, doc, row):
        view_name = _ensure_parent_view(sw_, doc, row,
                                        callers.get("solidworks_drawing_create_projected_view"))
        if view_name:
            _prep_section_line(sw_, doc, row, view_name)

    def precond(_sw, _doc, before, row):
        counts = before.get("selection_counts") or {}
        detail = {
            "selection_counts_by_mark": counts,
            "any_selection": any(v > 0 for v in counts.values()),
            "official": ("★ `CreateSectionViewAt5` 的 Remarks 逐字：Before calling this method, "
                         "select the section line or lines ⇒ ★ **本工具不代选**"),
        }
        return detail, bool(detail["any_selection"])

    return _run_case(sw, callers, label="T2", tool="solidworks_drawing_create_section_view",
                     kwargs={"x_mm": SECTION_X_MM, "y_mm": SECTION_Y_MM},
                     setup=setup, precond=precond, deltas=("views", "selection"),
                     call_even_if_pre_fails=True)


def case_t3_detail(sw, callers):

    ★ 实参逐字按 spec.params：`x_mm` · `y_mm` · `scale`(比例**分子**，分母固定 1.0)

def case_t8_center_marks(sw, callers):
    """★ T8 中心符号线：`solidworks_drawing_insert_center_marks`（`IView::AutoInsertCenterMarks2`）。

    ★ 实参逐字按 spec.params：`requirements`(list, 必填) —— ★ **`list` 型入参本仓无先例**
      （`recon/cand-C.md` §3.2 逐字登记过），★ 本探针按能力层读到的**数据形状**造：
      `{"id":…, "view":…, "count":…, "targets":[…]}`（★ `drawing_workflow.py:1952-1954`）。
    ★ 前置（★ 从能力层源码读到，官方页没写）：它拿 `drawing_model.GetPathName()`，
      ★ 图纸**没存盘**就回 `DRAWING_CENTER_MARK_DRAWING_SAVE_REQUIRED` ⇒ ★ 本格先存盘再调。
    ★ ★ spec 的 docstring 逐字写着：单点入口 `InsertCenterMark3` **已实测为阴性**，
      ★ 本工具走的是**批量**那条路、★ **尚未单独真机验证** ⇒ ★ 本格是它的**第一次**验证。
    ★ 判据：**中心符号读数增量 ≥ 1**（★ `IView::GetCenterMarkCount2`）。
    """
    def setup(sw_, doc, row):
        view_name = _ensure_parent_view(sw_, doc, row,
                                        callers.get("solidworks_drawing_create_projected_view"))
        try:
            row["saved_for_center"] = bool(C.save_document(doc, str(CENTER_DRAWING_PATH)))
        except Exception as exc:                                   # noqa: BLE001
            row["saved_for_center"] = "%s: %s" % (type(exc).__name__, str(exc)[:200])

    def precond(_sw, _doc, before, row):
        detail = {
            "parent_view": row.get("setup_parent_view"),

                         "（空路径 ⇒ `DRAWING_CENTER_MARK_DRAWING_SAVE_REQUIRED`）⇒ 如实标注"),
        }
        return detail, bool(detail["parent_view"] and detail["drawing_saved"]
                            and detail["drawing_path_nonempty"])

    requirements = [{"id": "probe-front-holes", "view": "front", "count": 1,
                     "targets": ["holes"]}]

                     kwargs={"requirements": requirements},

def _selected(filters, label) -> bool:
    return (not filters) or any(f.lower() in label.lower() for f in filters)


def main() -> int:
    """★ 跑全部（或筛选后的）格子 ⇒ 写 `output/drawing-tools-e2e-evidence.json`。"""
    filters = [a for a in sys.argv[1:] if not a.startswith("-")]
    ev = {
        "at": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "_": ("batch75 的 8 条工程图工具**经工具层**端到端（ToolSpec → tool_registry → "
              "scripts.sw_drawing_expose）。★ 判据一律用**读数增量**；★ 三态："
              "built / precondition_not_held / true_negative；★ harness_error 是探针自身问题，"
              "**不算**能力判据。"),
        "spec": "specs/202-user-coverage-gaps",
        "filters": filters,
        "expected_tools": list(EXPECTED_TOOLS),
        "part_path": str(PART_PATH),
        "drawing_template": DRAWING_TEMPLATE,
        "cases": [],
    }
    specs, spec_info = _collect_expose_specs()
    ev["tool_layer"] = {
        "spec_entry": spec_info,
        "registry_entry": ("mcp-server/tool_registry.py::build_input_model + "
                           "make_tool_function（★ 与 register() 内部同一条路）"),
        "resolvers_entry": "mcp-server/register_capabilities.py::RESOLVERS（原表，drawing_model → resolve_model）",
        "run_locked_stand_in": ("本文件 _run_locked（★ 照 server.py:1031 的形状：跑 op() + "
                                "异常变 status=error 的 JSON 串 + JSON 序列化）"),
        "why_not_import_server": ("★ server.py 在 import 期就 register_capabilities(整个工具面)，"
                                  "★ 副作用超出探针范围 ⇒ 不 import"),
    }
    callers = _tool_callers(specs)
    ev["tool_layer"]["callers_built"] = sorted(callers)

    sw = None
    part = None
    try:
        sw, _model, _meta = C.connect_solidworks(wait_seconds=30, return_metadata=True)
        try:
            sw.CommandInProgress = True
            ev["command_in_progress"] = True
        except Exception as exc:                                   # noqa: BLE001
            ev["command_in_progress"] = "%s: %s" % (type(exc).__name__, str(exc)[:120])
        ev["part_build"] = {}
        part = _build_part(sw, ev["part_build"])
        ev["part_build"]["part_ready"] = part is not None
        ev["part_build"]["saved_exists"] = PART_PATH.is_file()
        for label, desc, fn in CASES:
            if not _selected(filters, label):
                continue
            t0 = time.time()
            try:
                case = fn(sw, callers)
            except Exception as exc:                               # noqa: BLE001
                case = {"label": label, "ok": False, "state": "harness_error",
                        "harness_error": "%s: %s" % (type(exc).__name__, str(exc)[:240])}
            case.setdefault("label", label)
            case["desc"] = desc
            case["total_seconds"] = round(time.time() - t0, 2)
            ev["cases"].append(case)
            print("  %-4s %-42s state=%-22s pre=%-6s %ss"
                  % (label, desc[:42], case.get("state"), case.get("preconditions_held"),
                     case.get("total_seconds")))
            for key in ("tool_payload_ok", "tool_error", "tool_call_error", "deltas",
                        "signals", "sheet_readback_changed", "capability_raw_status",
                        "capability_raw_error_code", "tool_ok_vs_readback", "blocked",
                        "harness_error"):
                if key in case:
                    print("        %-26s %s" % (key, str(case[key])[:170]))
    except Exception as exc:                                       # noqa: BLE001
        ev["fatal"] = "%s: %s" % (type(exc).__name__, str(exc)[:300])
    finally:
        #: ★ 铁律 2 收尾：复位 `CommandInProgress`、关掉**自己开的**文档
        if sw is not None:
            try:
                sw.CommandInProgress = False
            except Exception:                                      # noqa: BLE001
                pass
            if part is not None:
                _close(sw, part)
            ev["command_in_progress_reset"] = True

    states = ("built", "precondition_not_held", "true_negative", "harness_error")
    ev["summary"] = {s: sum(1 for c in ev["cases"] if c.get("state") == s) for s in states}
    ev["summary"]["total"] = len(ev["cases"])
    ev["summary"]["built_with_preconditions_held"] = sum(
        1 for c in ev["cases"] if c.get("state") == "built" and c.get("preconditions_held"))
    ev["summary"]["tool_ok_disagreements"] = [
        {"label": c.get("label"), **c["tool_ok_vs_readback"]}
        for c in ev["cases"]
        if isinstance(c.get("tool_ok_vs_readback"), dict)
        and c["tool_ok_vs_readback"].get("agree") is False]
    OUT_JSON.parent.mkdir(parents=True, exist_ok=True)
    OUT_JSON.write_text(json.dumps(ev, ensure_ascii=False, indent=1, default=str) + "\n",
                        encoding="utf-8")
    print("  判读：%s" % json.dumps(ev["summary"], ensure_ascii=False))
    print("  ⚠ ★ `state=precondition_not_held` 的格子**不能**当「能力不成」的证据。")
    print("  ⚠ ★ `tool_ok_disagreements` = 工具自报的 ok 与独立读数**不一致**的格子。")
    print("  产物：%s" % OUT_JSON)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
