def _result(payload, response_format) -> str:
    """★ 镜像 `server.py:1001-1014`。"""
    if response_format == ResponseFormat.JSON:
        return json.dumps(payload, ensure_ascii=False, indent=2)
    lines = ["# %s" % payload.get("status", "result")]
    for key, value in payload.items():
        if key == "status":
            continue
        if isinstance(value, (dict, list)):
            lines.append("- **%s**:\n```json\n%s\n```"
                         % (key, json.dumps(value, ensure_ascii=False, indent=2)))
        else:
            lines.append("- **%s**: %s" % (key, value))
    return "\n".join(lines)


def _tool_error(exc: Exception, response_format=ResponseFormat.JSON) -> str:
    """★ 镜像 `server.py:1017-1028`。"""
    return _result({
        "status": "error",
        "error_type": type(exc).__name__,
        "message": str(exc),
        "suggestion": (
            "Confirm SolidWorks is installed/running, the active document is correct, "
            "components are resolved, and file paths are absolute Windows paths."
        ),
    }, response_format)


def _run_locked(operation, response_format, load_automation: bool = True):
    """★ 镜像 `server.py:1031-1042`（★ 工具层就是用它把异常变成 `{"status":"error", …}` 字符串的）。

    ★ `load_automation` 在真 server 里加载一批 `scripts.*` 模块；★ 本探针需要的
      （`sw_connect` / `sw_part` / `sw_drawing_expose` / `sw_drawing_query`）**已在模块级 import**，
      ⇒ 这里只在首次调用时 import 一次 `scripts.sw_connect` 作为等价物，★ 并如实记进 evidence。
    """
    global _AUTOMATION_TOUCHED
    with _sw_lock:
        try:
            if load_automation and not _AUTOMATION_TOUCHED:
                import importlib

                importlib.import_module("scripts.sw_connect")
                _AUTOMATION_TOUCHED = True
            _coinitialize()
            with redirect_stdout(sys.stderr):
                payload = operation()
            return _result(payload, response_format)
        except Exception as exc:                                        # noqa: BLE001
            return _tool_error(exc, response_format)

# ── 取声明 + 造可调函数（★ 工具层的真实入口）─────────────────────────────
def load_tool_functions():
    """★ 用 `all_specs()` 取 8 条声明 → `build_input_model` + `make_tool_function` ⇒ 真可调函数。

    ★ 取不到就退回**直接**读 `tool_specs_batch75.SPECS_BATCH75`（并**如实记下退回了**），
      ★ 不静默换来源。
    """
    from tool_registry import build_input_model, make_tool_function
    from register_capabilities import RESOLVERS

    notes: list[str] = []
    specs = None
    try:
        from register_capabilities import all_specs

        specs = [s for s in all_specs() if s.name in TARGET_TOOLS]
        notes.append("spec 来源 = register_capabilities.all_specs()")
    except Exception as exc:                                            # noqa: BLE001
        notes.append("all_specs() 取失败（%s: %s）⇒ 退回直接读 tool_specs_batch75.SPECS_BATCH75"
                     % (type(exc).__name__, str(exc)[:180]))
        import tool_specs_batch75

        specs = [s for s in tool_specs_batch75.SPECS_BATCH75 if s.name in TARGET_TOOLS]

    base = _base_input_cls()
    entries: dict[str, dict] = {}
    for spec in specs:
        model = build_input_model(spec, base, ResponseFormat)
        entries[spec.name] = {
            "spec": spec,
            "input_model": model,
            "fn": make_tool_function(spec, model, _run_locked, None, RESOLVERS),
        }
    missing = [n for n in TARGET_TOOLS if n not in entries]
    return entries, missing, notes, specs


def contract_report(spec) -> dict:
    """★ **静态**把 ToolSpec 的声明与**能力层真实签名**对差（★ 不碰 COM、不调工具）。

    ★ 这是本探针的第一现场：`inspect.signature` 读的是能力层**源码**里的真实签名 ⇒
      「声明的参数名能力层收不进来」在**跑真机之前**就能指认。
    ★ 取不到签名就写「读不到」，★ **不猜、不编**。
    """
    from tool_registry import effective_inject

    rep: dict = {
        "tool": spec.name,
        "title": spec.title,
        "module": spec.module,
        "function": spec.function,
        "inject": effective_inject(spec),
        "declared_params": [
            {"name": p.name, "kind": p.kind.__name__, "required": bool(p.is_required)}
            for p in spec.params
        ],
        "omitted_params": list(spec.omitted_params),
        "param_alias": dict(spec.param_alias),
        "has_derived_args": spec.derived_args is not None,
        "has_pre_call": spec.pre_call is not None,
        "load_automation": bool(spec.load_automation),
    }
    target = getattr(DX, spec.function, None)
    if target is None:
        rep["capability_readable"] = False
        rep["note"] = "★ 能力层取不到 %s.%s ⇒ 读不到，不猜" % (spec.module, spec.function)
        return rep
    rep["capability_readable"] = True
    try:
        sig = inspect.signature(target)
    except Exception as exc:                                            # noqa: BLE001
        rep["signature_readable"] = False
        rep["note"] = "★ 取签名失败：%s: %s" % (type(exc).__name__, str(exc)[:140])
        return rep
    rep["signature_readable"] = True
    injected = effective_inject(spec)
    caps = {n: (p.default is inspect.Parameter.empty)
            for n, p in sig.parameters.items()}
    passed = [p.name for p in spec.params]
    rep["capability_params"] = sorted(caps)
    rep["capability_required"] = sorted(n for n, req in caps.items() if req)
    rep["tool_layer_would_pass"] = sorted(passed)
    rep["accepted_by_capability"] = sorted(n for n in passed if n in caps)
    rep["NOT_accepted_by_capability"] = sorted(n for n in passed if n not in caps)
    rep["required_by_capability_but_UNdeclared"] = sorted(
        n for n, req in caps.items() if req and n not in passed and n != injected)
    rep["optional_in_capability_but_UNdeclared"] = sorted(
        n for n, req in caps.items() if not req and n not in passed and n != injected)
    rep["contract_ok"] = (not rep["NOT_accepted_by_capability"]
                          and not rep["required_by_capability_but_UNdeclared"])
    rep["call_shape"] = "make_tool_function 会把 %r 与声明的每个参数**同名**当关键字实参传进去" % injected
    return rep

def _activate_doc(sw, doc):
    """★ 激活文档（`ISldWorks.ActivateDoc3`，第 4 参是 **by-ref Int32**）。

    ★★ **为什么必须有这一步**：`register_capabilities.resolve_model` 注入的是 **`sw.ActiveDoc`**
      （`register_capabilities.py:53-60`，底层 `sw_connect._read_active_document` =
      `get_com_member(sw, "ActiveDoc")`）⇒ ★ 工具层拿到的 `drawing_model` **只可能是当前活动文档**。
    ★ 形态抄 `scripts/sw_assembly.py:108-110` 与 `scripts/sw_export.py:251`（本仓已用过）。
    ★ 判据 = **回读** `sw.ActiveDoc` 的标题（★ 返回值不可信，同事实 8 的口径）。
    """
    try:
        title = str(_get(doc, "GetTitle") or "")
    except Exception as exc:                                            # noqa: BLE001
        return False, "取 GetTitle 失败：%s: %s" % (type(exc).__name__, str(exc)[:100])
    if not title:
        return False, "GetTitle 回空串"
    try:
        errors = C.VARIANT(C.pythoncom.VT_BYREF | C.pythoncom.VT_I4, 0)
    except Exception as exc:                                            # noqa: BLE001
        return False, "造 by-ref VARIANT 失败：%s: %s" % (type(exc).__name__, str(exc)[:100])
    try:
        _get(sw, "ActivateDoc3", title, False, 0, errors)
    except Exception as exc:                                            # noqa: BLE001
        try:
            _get(sw, "ActivateDoc2", title, False, errors)
        except Exception as exc2:                                       # noqa: BLE001
            return False, "ActivateDoc3 %s: %s / ActivateDoc2 %s: %s" % (
                type(exc).__name__, str(exc)[:90], type(exc2).__name__, str(exc2)[:90])
    try:
        active = _get(sw, "ActiveDoc")
        active_title = str(_get(active, "GetTitle") or "") if active is not None else ""
    except Exception as exc:                                            # noqa: BLE001
        return False, "回读 ActiveDoc 失败：%s: %s" % (type(exc).__name__, str(exc)[:100])
    return active_title == title, {"want": title, "active_doc_title": active_title}

    """★ 视图链 `GetFirstView()` → `GetNextView()`（★ 含"图纸本身"那个伪视图）；有防死循环上限。"""
    out: list = []
    try:
        v = _get(doc, "GetFirstView")
    except Exception as exc:                                            # noqa: BLE001
        return [], "%s: %s" % (type(exc).__name__, str(exc)[:120])
    n = 0
    while v is not None and n < 500:
        n += 1
        out.append(v)
        try:
            v = _get(v, "GetNextView")
        except Exception:                                               # noqa: BLE001
            break
    return out, None


def _real_views(doc):
    """★ **真视图**（排除"图纸本身"那个伪视图）—— 名单来自**已真机验证过**的 `DQ.list_views`。"""
    try:
        names = [str((r or {}).get("name") or "") for r in (DQ.list_views(doc) or [])]
    except Exception as exc:                                            # noqa: BLE001
        return [], "取视图清单失败：%s: %s" % (type(exc).__name__, str(exc)[:120])
    chain, err = _chain(doc)
    if err:
        return [], err
    return [v for v in chain if _vname(v) in names], None

def _inside_points(bbox):
    """★ 由视图包围盒推出候选拾取点（都在视图内，从中心往外）。"""
    xmin, ymin, xmax, ymax = bbox[:4]
    xm, ym = (xmin + xmax) / 2.0, (ymin + ymax) / 2.0
    w, h = (xmax - xmin), (ymax - ymin)
    return [
        (xm, ym),
        (xmin + 0.05 * w, ym), (xmax - 0.05 * w, ym),
        (xm, ymin + 0.05 * h), (xm, ymax - 0.05 * h),
        (xmin + 0.02 * w, ymin + 0.02 * h), (xmax - 0.02 * w, ymax - 0.02 * h),
        (xmin + 0.02 * w, ymax - 0.02 * h), (xmax - 0.02 * w, ymin + 0.02 * h),
    ]


def _pick(doc, kind, x, y, mark=0, append=False):
    """★ 按点选（`IModelDocExtension.SelectByID2`，9 参、第 7 个是 `Mark`）。"""
    try:
        _get(_get(doc, "Extension"), "SelectByID2", "", str(kind),
             float(x), float(y), float(PICK_Z_M), bool(append), int(mark),
             C.create_empty_dispatch_variant(), 0)
    except Exception as exc:                                            # noqa: BLE001
        return False, "%s: %s" % (type(exc).__name__, str(exc)[:120])
    return True, None


def _pick_any(doc, kind, points, mark=0, append=False):
    """★ 候选点上**逐个试**按点选，直到**回读计数**涨了 ⇒ `(命中点 | None, 计数, 错误)`。"""
    before = _sel_counts(doc)
    errs = []
    for (x, y) in points:
        _ok, err = _pick(doc, kind, x, y, mark=mark, append=append)
        if err:
            errs.append(err)
        after = _sel_counts(doc)
        if sum(v for v in after.values() if v > 0) > sum(v for v in before.values() if v > 0):
            return (round(x, 6), round(y, 6)), after, (errs or None)
    return None, _sel_counts(doc), (errs or None)


#: ★ 各族的**官方回读成员**（★ 全部零参；★ 判据不许只看特征树）。
_COUNTER_MEMBERS = (
    ("note", "GetNoteCount"),                   #: 注释（气泡也是 note）
    ("table", "GetTableAnnotationCount"),       #: 表格（孔表 / 修订表 / BOM）
    ("center", "GetCenterMarkCount2"),          #: 中心符号线
)


def _anno_totals(doc) -> dict:
    """★ 注解读数：**真视图**与**整条视图链**各算一份（判据取"任一份涨了"）。

    ⚠ 为什么两份：★ 有些注解挂在**图纸**上而不挂在某个视图里 ⇒ 只数真视图会漏；
      整条链又把"图纸本身"算进来 ⇒ ★ 两份都报出来，让人看得见是哪一份涨的。
    """
    real, _e1 = _real_views(doc)
    chain, _e2 = _chain(doc)
    state: dict = {}
    for label, views in (("real", real), ("all", chain)):
        rows = []
        for v in views:
            row = {"name": _vname(v)}
            for key, member in _COUNTER_MEMBERS:
                try:
                    row[key] = int(_get(v, member) or 0)
                except Exception:                                       # noqa: BLE001
                    row[key] = None
            rows.append(row)
        totals = {}
        for key, _m in _COUNTER_MEMBERS:
            vals = [r[key] for r in rows if isinstance(r.get(key), int)]
            totals[key] = sum(vals) if vals else None
        state[label] = {"count": len(rows), "totals": totals, "per_view": rows}
    return state


def _delta(before, after, key, bucket="all"):
    """★ 增量（★ 任一份量不到就回 None ⇒ 判据降级，不猜 0）。"""
    try:
        b = before[bucket]["totals"].get(key)
        a = after[bucket]["totals"].get(key)
    except Exception:                                                   # noqa: BLE001
        return None
    if b is None or a is None:
        return None
    return int(a) - int(b)


def _enum(enum_name: str, member: str):
    """★ **在官方枚举页里现场取枚举值** ⇒ `(value|None, 来源或原因)`；读不到就回 None 并说明。

    ★ 解析规则只认页里那张 `| Member | Value |` 表（照 `prove_drawing_batch._enum`）。
    """
    key = (enum_name, member)
    if key in _ENUM_CACHE:
        return _ENUM_CACHE[key]
    page = SWAPI_ENUMS / ("%s.md" % enum_name)
    val: int | None = None
    note: str | None = None
    if not page.is_file():
        note = "★ 官方枚举页不在本机：%s" % page
    else:
        try:
            text = page.read_text(encoding="utf-8", errors="replace")
        except OSError as exc:
            text, note = "", "★ 读官方枚举页失败：%s" % str(exc)[:120]
        if text:
            for line in text.splitlines():
                if not line.startswith("|"):
                    continue
                cells = [c.strip() for c in line.strip().strip("|").split("|")]
                if len(cells) < 2 or cells[0] != member:
                    continue
                raw = cells[1]
                m = re.match(r"^0x([0-9a-fA-F]+)", raw) or re.match(r"^(-?\d+)", raw)
                if m:
                    val = int(m.group(1), 16) if raw.lower().startswith("0x") else int(m.group(1))
                    note = str(page)
                else:
                    note = "★ %s 的 Value 列不是数字（%r）⇒ 不解析、不猜" % (member, raw[:60])
                break
            else:
                note = "★ 官方枚举页 %s 里没有成员 %s ⇒ 不猜" % (page.name, member)
    _ENUM_CACHE[key] = (val, note)
    return val, note


def _ev(row: dict, enum_name: str, member: str, label: str = ""):
    """★ 取枚举值；取不到就**记进 row 并回 None**（调用方据此跳过，不硬凑）。"""
    val, note = _enum(enum_name, member)
    if val is None:
        row.setdefault("enum_missing", []).append(
            "%s.%s%s ⇒ %s" % (enum_name, member, ("（%s）" % label) if label else "", note or ""))
    return val


def _find_files(patterns, roots):
    """★ 在若干根目录下按扩展名找文件（用到才找；找不到如实回空）。"""
    hits: list[str] = []
    for root in roots:
        try:
            base = Path(root)
            if not base.is_dir():
                continue
            for pat in patterns:
                hits.extend(sorted(str(p) for p in base.rglob(pat)))
        except OSError:
            continue
    return hits

def _build_part(sw, row: dict):
    """★ 建一块**带孔**的板并**存盘**。

    ★★ 为什么必须存盘：官方 `CreateDrawViewFromModelView3` 的 Remarks 逐字要求 `ModelName` 是
      **完整路径**且**含 `.sldprt`/`.sldasm` 扩展名** ⇒ `create_projected_view` 的前置也是它。
    ★ 板上打一个孔：好让"孔表 / 中心符号线"两族有**圆边/圆面**可用。
    """
    doc = C.new_document(sw, "part")
    try:
        P.start_sketch(doc, plane_name="Front Plane")
        _get(_get(doc, "SketchManager"), "CreateCornerRectangle",
             -0.04, -0.03, 0.0, 0.04, 0.03, 0.0)
        P.end_sketch(doc)
        names = [n for n in (P.feature_names(doc) or [])
                 if n.startswith("\u8349\u56fe") or n.lower().startswith("sketch")]
        if not names:
            row["extrude_err"] = "找不到草图特征名（feature_names 没回草图）"
            return None
        P.extrude_boss(doc, names[-1], 0.01)
        row["extrude_sketch"] = names[-1]
    except Exception as exc:                                            # noqa: BLE001
        row["extrude_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
        return None
    try:
        from scripts import sw_hole_features as HF

        rep = HF.create_hole_wizard_at_selection_report(doc, "+z", "M8")
        row["hole"] = {k: rep.get(k) for k in ("ok", "selection_mark") if k in rep}
    except Exception as exc:                                            # noqa: BLE001
        #: ★ 孔不成**不阻断**：只是那两族的前置会更弱 ⇒ 如实记下来
        row["hole_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    try:
        row["saved"] = bool(C.save_document(doc, str(PART_PATH)))
    except Exception as exc:                                            # noqa: BLE001
        row["save_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
        row["saved"] = False
    return doc if row.get("saved") else None

def _resolve_model_view_name(part_doc, info: dict | None = None):
    """★ **问模型自己**要一个合法视图名（★ 不硬编码 —— 名字会被 UI 本地化）。

    ★ 依据（事实 494）：官方 `CreateDrawViewFromModelView3` 的 Remarks 要求 `ViewName`
      "**exactly match**" 模型视图名，而它给的例子 `"*Front"` 是**英文界面**的取值
      ⇒ 本机中文界面是 `*前视` ⇒ ★ 照抄 `*Front` 必回 `None` 且**无任何报错**。
    ★ 正解：`IModelDoc2.GetModelViewNames()`（零参、未废弃）。
      ⚠ 同族的 `GetModelViewCount()` 官方标 **Obsolete** ⇒ **不用它**。
    """
    try:
        names = [str(x) for x in (_get(part_doc, "GetModelViewNames") or [])]
    except Exception as exc:                                            # noqa: BLE001
        names = []
        if info is not None:
            info["view_names_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    if info is not None:
        info["model_view_names"] = names
    return (names[0] if names else None)


def _ensure_parent_view(ctx: dict, row: dict):
    """★ 脚手架：给图上放一个**父视图**（下游四族的前置）。

    ⚠⚠ **这是内部调用、不是被测对象**：`create_projected_view` 自己就是 8 条里的一条，
      ★ 所以父视图**不能**靠它建（否则那一格的成败与脚手架纠缠）。
      ⇒ 这里直调能力层 `DX.create_projected_view`，**只传它真实签名收得下的参数**，
      ★ 并在 evidence 里标 `internal_call=True`。
    ★ 它走 `add_view` → `CreateDrawViewFromModelView3`，视图名由 `_create_model_view` 解析
      （事实 494）—— ★ 本探针**再显式传**本地化真名，双保险。
    """
    if ctx.get("view_name"):
        return ctx["view_name"], {"reused": True}
    drawing, part_doc = ctx.get("drawing"), ctx.get("part_doc")
    if drawing is None or part_doc is None:
        row["scaffolding_err"] = "★ 零件或工程图不在 ⇒ 建不出父视图"
        return None, None
    vname = _resolve_model_view_name(part_doc, row)
    row["parent_view_view_name"] = vname
    if not vname:
        row["scaffolding_err"] = "★ 前置未成立：GetModelViewNames 回空 ⇒ 问不到合法视图名"
        return None, None
    before = _view_count(drawing)
    try:
        res = DX.create_projected_view(drawing, str(PART_PATH), view_name=str(vname))
    except Exception as exc:                                            # noqa: BLE001
        row["scaffolding_err"] = "直调能力层失败：%s: %s" % (type(exc).__name__, str(exc)[:200])
        return None, None
    views, _e = _real_views(drawing)
    names = [_vname(v) for v in views]
    row["scaffolding"] = {
        "internal_call": True,
        "note": "★ 内部函数，参数契约与工具层不同（脚手架，不是被测对象）",
        "call": "DX.create_projected_view(drawing, part_path, view_name=<本地化真名>)",
        "return_ok": (res or {}).get("ok") if isinstance(res, dict) else None,
        "preconditions_held": (res or {}).get("preconditions_held") if isinstance(res, dict) else None,
        "views_before": before, "views_after": _view_count(drawing), "view_names": names,
    }
    if not names:
        row["scaffolding_err"] = "★ 父视图没建成（视图清单为空）⇒ 下游四族前置未成立"
        return None, None
    ctx["view_name"] = names[0]
    return ctx["view_name"], row["scaffolding"]


# ── 调用工具层 ────────────────────────────────────────────────────────
def _call_and_parse(entry: dict, kwargs: dict, row: dict):
    """★ 按 ToolSpec 声明构造输入模型并调**工具层**函数 ⇒ `(解析后的 dict | None, 异常 | None)`。

    ★ 注意异常分两层：① 造输入模型（`extra="forbid"` ⇒ 未声明的参数会在这里被 pydantic 拒）
      ② 调函数（异常被 `_run_locked` 收成 `{"status":"error", …}` 字符串）★ 两层都记。
    """
    declared = [p.name for p in entry["spec"].params]
    row["kwargs_sent"] = {
        k: (v if isinstance(v, (str, int, float, bool)) else "★ 结构型，见 tool_return")
        for k, v in kwargs.items()
    }
    not_declared = sorted(set(kwargs) - set(declared))
    if not_declared:
        row["kwargs_not_declared_by_spec"] = not_declared
    try:
        inst = entry["input_model"](**kwargs)
    except Exception as exc:                                            # noqa: BLE001
        return None, "★ 造输入模型失败：%s: %s" % (type(exc).__name__, str(exc)[:240])
    try:
        text = entry["fn"](inst)
    except Exception as exc:                                            # noqa: BLE001
        return None, "★ 调工具层失败：%s: %s" % (type(exc).__name__, str(exc)[:240])
    row["tool_return_chars"] = len(text or "")
    try:
        return json.loads(text), None
    except Exception as exc:                                            # noqa: BLE001
        row["tool_return_not_json"] = (text or "")[:400]
        return None, "★ 返回值不是 JSON：%s" % str(exc)[:120]


def _contract_typeerror(payload):
    """★ 返回值是不是「**工具层按声明传参 ⇒ 能力层签名收不进**」的 TypeError。"""
    if not isinstance(payload, dict) or payload.get("status") != "error":
        return None
    if payload.get("error_type") != "TypeError":
        return None
    msg = str(payload.get("message") or "")
    if ("unexpected keyword argument" in msg
            or ("keyword argument" in msg and "required" in msg)):
        return msg
    return None


def _finish(row: dict, verdict: str, ok: bool, note: str = "") -> dict:
    """★ 统一收口：判据只认 `verdict` + `ok`，★ 前置单独记（事实 496）。"""
    row["verdict"] = verdict
    row["ok"] = bool(ok)
    if note:
        row["note"] = note
    return row


# ── 八格（★ 每格一格；★ 每格 try/except 在 main 里）────────────────────────
def case_projected_view(entry, ctx, row):
    """★ ① `solidworks_drawing_create_projected_view`（声明：`part_path` + `projection`）。

    ★ 官方 `CreateDrawViewFromModelView3` 硬前置：`ModelName` = **含 `.sldprt` 的完整路径**。
    ★ 判据：**视图数 +1** 且返回值 `ok=True`。
    """
    drawing = ctx.get("drawing")
    if drawing is None or ctx.get("part_doc") is None:
        return _finish(row, "precondition_not_met", False,
                       "★ 前置未成立：没有已存盘的零件 / 没有新建的工程图 ⇒ 本格不作判据")
    if not PART_PATH.is_file():
        return _finish(row, "precondition_not_met", False,
                       "★ 前置未成立：%s 不在 ⇒ ModelName 的硬前置不成立" % PART_PATH)
    ok_act, detail = _activate_doc(ctx["sw"], drawing)
    row["activated_drawing"] = detail
    if not ok_act:
        return _finish(row, "precondition_not_met", False,
                       "★ 前置未成立：激活不了目标工程图 ⇒ resolver 注入不到它")
    before = _view_count(drawing)
    row["readback_before"] = {"view_count": before}
    payload, err = _call_and_parse(entry, {
        "part_path": str(PART_PATH),
        "projection": "first_angle",                       # ★ 按 spec 的描述原样给，不改成别的值
    }, row)
    row["tool_return"] = payload
    if err:
        row["call_err"] = err
    row["readback_after"] = {"view_count": _view_count(drawing)}
    row["view_delta"] = (None if (before is None or row["readback_after"]["view_count"] is None)
                         else row["readback_after"]["view_count"] - before)
    cterr = _contract_typeerror(payload)
    if cterr:
        row["contract_typeerror"] = cterr
        # 分离诊断：能力层自己（用**它真实签名**）成不成？
        _diag_projected_view(ctx, row)
        return _finish(row, "tool_layer_contract_mismatch", False,
                       "★ 按 ToolSpec 声明传参 ⇒ 能力层签名收不进 ⇒ **不是能力不成、也不是前置问题**"
                       "（分离诊断见 internal_diagnostic）")
    if err:
        return _finish(row, "probe_error", False, err)
    if bool((payload or {}).get("ok")) and (row["view_delta"] or 0) >= 1:
        return _finish(row, "built", True, "视图数 +1 且返回值 ok=True")
    if (payload or {}).get("preconditions_held") is False:
        return _finish(row, "precondition_not_met", False,
                       "★ 工具层自己报 preconditions_held=False ⇒ 不能据此判「投影视图不成」")
    return _finish(row, "true_negative", False, "调用发生了，但视图数没涨 / ok 不为真")


def _diag_projected_view(ctx, row):
    """★ 分离诊断：能力层 `DX.create_projected_view` 用**它自己的签名**直调一次。

    ★ 标注 `internal_call=True`：**内部函数，参数契约与工具层不同** ⇒ 这一格只回答
      「工具层坏了，还是能力层也坏了」。
    """
    drawing, part_doc = ctx.get("drawing"), ctx.get("part_doc")
    diag: dict = {"internal_call": True,

def _prep_section_line(ctx: dict, row: dict):
    """★ 预备剖面线 —— ★ 照官方例子的**调用序**（`prove_drawing_batch._prep_section_line`）。

    官方例子 `Create_Section_View_at_Specified_Location_Example_VB`：
      `ActivateView(视图名)` → `SelectByID2(视图名,"DRAWINGVIEW")` → `ClearSelection2`
      → `CreateLine2(...)`（★ 剖面线就是**活动视图里画的一条草图线**）→ 才调 `CreateSectionViewAt5`。
    ★ 判据 = **回读选区计数**（★ 不信 `SelectByID2` 的返回值）。
    """
    doc, name = ctx["drawing"], ctx["view_name"]
    row["activated_view"] = _activate_view(doc, name)
    _ok, counts = _select_view(doc, name, mark=0)
    row["sel_view_counts"] = counts
    _clear(doc)
    views, _e = _real_views(doc)
    view = next((v for v in views if _vname(v) == name), None)
    bbox = _outline(view) if view is not None else None
    row["view_outline_m"] = [round(v, 6) for v in bbox] if bbox else None
    if not bbox:
        return False, "取不到视图包围盒（IView.GetOutline 没回 4 个数）"
    y_mid = (bbox[1] + bbox[3]) / 2.0
    p1 = (bbox[0] + 0.1 * (bbox[2] - bbox[0]), y_mid)
    p2 = (bbox[2] - 0.1 * (bbox[2] - bbox[0]), y_mid)
    try:
        _get(_get(doc, "SketchManager"), "CreateLine2",
             float(p1[0]), float(p1[1]), 0.0, float(p2[0]), float(p2[1]), 0.0)
        row["line_drawn_m"] = [round(p1[0], 6), round(p1[1], 6), round(p2[0], 6), round(p2[1], 6)]
    except Exception as exc:                                            # noqa: BLE001
        return False, "CreateLine2 失败：%s: %s" % (type(exc).__name__, str(exc)[:140])
    counts = _sel_counts(doc)
    row["sel_by_mark_before_call"] = counts
    held = sum(v for v in counts.values() if v > 0) > 0
    return held, ("选区为空" if not held else "")

def _prep_detail_circle(ctx, row) -> bool:
    """★ 在活动视图里画一个**局部圆**（官方例子的调用序：`ActivateView` → `CreateCircle`）。"""
    doc, name = ctx["drawing"], ctx["view_name"]
    row["activated_view"] = _activate_view(doc, name)
    _clear(doc)
    views, _e = _real_views(doc)
    view = next((v for v in views if _vname(v) == name), None)
    bbox = _outline(view) if view is not None else None
    row["view_outline_m"] = [round(v, 6) for v in bbox] if bbox else None
    if not bbox:
        row["detail_circle_err"] = "取不到视图包围盒 ⇒ 画不出局部圆"
        return False
    cx, cy = (bbox[0] + bbox[2]) / 2.0, (bbox[1] + bbox[3]) / 2.0
    r = 0.2 * min(bbox[2] - bbox[0], bbox[3] - bbox[1])
    if r <= 0:
        row["detail_circle_err"] = "视图包围盒退化（r=%r）⇒ 画不出局部圆" % r
        return False
    try:
        _get(_get(doc, "SketchManager"), "CreateCircle",
             float(cx), float(cy), 0.0, float(cx + r), float(cy), 0.0)
    except Exception as exc:                                            # noqa: BLE001
        row["detail_circle_err"] = "CreateCircle 失败：%s: %s" % (type(exc).__name__, str(exc)[:140])
        return False
    row["detail_circle"] = {"center_m": [round(cx, 6), round(cy, 6)], "r_m": round(r, 6)}
    return True

                       "★ 前置未成立（%s）⇒ **不能**据此判「孔表不成」" % why)
    return _finish(row, "true_negative", False, "前置成立、调用发生了，但表注解数没涨")


def _prep_hole_table_marks(ctx, row):
    """★ 按官方 mark 表选「顶点 mark 1」+「面 mark 2」（★ 单变量：先不选 4/8 那个最小前置）。"""
    doc, name = ctx["drawing"], ctx["view_name"]
    row["activated_view"] = _activate_view(doc, name)
    _clear(doc)
    views, _e = _real_views(doc)
    view = next((v for v in views if _vname(v) == name), None)
    bbox = _outline(view) if view is not None else None
    row["view_outline_m"] = [round(v, 6) for v in bbox] if bbox else None
    if not bbox:
        return False, "取不到视图包围盒 ⇒ 无法推候选拾取点"
    pts = _inside_points(bbox)
    v_hit, _c1, e1 = _pick_any(doc, "VERTEX", pts, mark=1)
    f_hit, counts, e2 = _pick_any(doc, "FACE", pts, mark=2, append=True)
    row.update({"vertex_pick": v_hit, "face_pick": f_hit,
                "sel_by_mark_before_call": counts,
                "pick_errs": {"vertex": e1, "face": e2}})
    held = bool(v_hit is not None and f_hit is not None
                and counts.get("1", 0) >= 1 and counts.get("2", 0) >= 1)
    return held, ("官方 mark 表要的 1/2 没同时在选区里" if not held else "")

def case_revision_table(entry, ctx, row):
    """★ ⑦ `solidworks_drawing_create_revision_table`（声明：`x_mm`/`y_mm`/`table_type`/`row`/`column`/`template`/`use_anchor_point`）。

    ★ 官方 `ISheet.InsertRevisionTable2` 的 Remarks：模板是 `.sldrevtbt`、模板与表**必须同型**。
    ★ 官方 **Returns 逐字**：**已有修订表时回 `null`** ⇒ ★ 那**是契约、不是失败**（由调用方解释）。
    ★ 判据：返回真 **且** `GetTableAnnotationCount` +1。
    """
    drawing = ctx.get("drawing")
    if drawing is None:
        return _finish(row, "precondition_not_met", False, "★ 前置未成立：没有工程图")
    ok_act, detail = _activate_doc(ctx["sw"], drawing)
    row["activated_drawing"] = detail
    if not ok_act:
        return _finish(row, "precondition_not_met", False, "★ 前置未成立：激活不了目标工程图")
    try:
        sheet = _get(drawing, "GetCurrentSheet")
    except Exception as exc:                                            # noqa: BLE001
        sheet = None
        row["sheet_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:120])
    row["preconditions_held"] = sheet is not None
    if sheet is None:
        return _finish(row, "precondition_not_met", False,
                       "★ 前置未成立：取不到当前图纸（ISheet.GetCurrentSheet 回 None）")
    tpl = ctx.get("revision_table_template") or _revision_table_template()
    row["revision_table_template"] = tpl
    if not tpl:
        return _finish(row, "precondition_not_met", False,
                       "★ 前置未成立：找不到 `.sldrevtbt` 模板 ⇒ 本格不作判据")
    before = _anno_totals(drawing)
    row["readback_before"] = {"table": before["all"]["totals"]["table"]}
    payload, err = _call_and_parse(entry, {
        "x_mm": 400.0, "y_mm": 200.0, "table_type": 1, "row": 0, "column": 0,
        "template": str(tpl), "use_anchor_point": False,
    }, row)
    row["tool_return"] = payload
    if err:
        row["call_err"] = err
    after = _anno_totals(drawing)
    row["readback_after"] = {"table": after["all"]["totals"]["table"]}
    row["table_delta"] = _delta(before, after, "table", "all")
    row["official_returns_note"] = "★ 官方 Returns 逐字：已有修订表时回 null ⇒ 那**是契约、不是失败**"
    cterr = _contract_typeerror(payload)
    if cterr:
        row["contract_typeerror"] = cterr
        _diag_revision_table(ctx, row, tpl)
        return _finish(row, "tool_layer_contract_mismatch", False,
                       "★ 按 ToolSpec 声明传参 ⇒ 能力层签名收不进 ⇒ **不是能力不成**"
                       "（分离诊断见 internal_diagnostic）")
    if err:
        return _finish(row, "probe_error", False, err)
    ret_ok = bool((payload or {}).get("ok")) if isinstance(payload, dict) else bool(payload)
    raw_none = (payload or {}).get("raw_is_none") if isinstance(payload, dict) else None
    row["raw_is_none"] = raw_none
    if ret_ok and (row["table_delta"] or 0) >= 1:
        return _finish(row, "built", True, "表注解数 +1 且返回值 ok=True")
    if raw_none is True:
        return _finish(row, "precondition_not_met", False,
                       "★ 返回 null：**按官方 Returns 这就是「已经有修订表」** ⇒ 该格不作判据")
    return _finish(row, "true_negative", False, "调用发生了，但表注解数没涨")

def _selected(filters, name) -> bool:
    """★ 命令行筛选（不传筛选就跑全部）。"""
    return (not filters) or any(f.lower() in name.lower() for f in filters)
