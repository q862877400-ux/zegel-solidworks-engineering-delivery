def _enum(enum_name: str, member: str):
    """★ **在官方枚举页里现场取枚举值** ⇒ `(value|None, 来源或原因)`（照抄 `prove_drawing_batch._enum`）。"""
    key = (enum_name, member)
    if key in _ENUM_CACHE:
        return _ENUM_CACHE[key]
    page = SWAPI_ENUMS / ("%s.md" % enum_name)
    val: int | None = None
    note: str | None = None
    if not page.is_file():
        note = "★ 官方枚举页不在本机：%s" % page
    else:
        try:
            text = page.read_text(encoding="utf-8", errors="replace")
        except OSError as exc:
            text, note = "", "★ 读官方枚举页失败：%s" % str(exc)[:120]
        if text:
            for line in text.splitlines():
                if not line.startswith("|"):
                    continue
                cells = [c.strip() for c in line.strip().strip("|").split("|")]
                if len(cells) < 2 or cells[0] != member:
                    continue
                raw = cells[1]
                m = re.match(r"^0x([0-9a-fA-F]+)", raw) or re.match(r"^(-?\d+)", raw)
                if m:
                    val = int(m.group(1), 16) if raw.lower().startswith("0x") else int(m.group(1))
                    note = str(page)
                else:
                    note = "★ %s 的 Value 列不是数字（%r）⇒ 不解析、不猜" % (member, raw[:60])
                break
            else:
                note = "★ 官方枚举页 %s 里没有成员 %s ⇒ 不猜" % (page.name, member)
    _ENUM_CACHE[key] = (val, note)
    return val, note


def _need_enum(row: dict, enum_name: str, member: str):
    """★ 取枚举；取不到就**记进 row 并回 None**（★ 调用方据此跳过该格，不硬凑一个数）。"""
    val, note = _enum(enum_name, member)
    if val is None:
        row.setdefault("enum_missing", []).append("%s.%s ⇒ %s" % (enum_name, member, note or ""))
    return val

def _chain(doc) -> list:
    """★ 视图链 `GetFirstView()` → `GetNextView()`（★ **含**"图纸本身"那个伪视图 —— 真机实测过）。"""
    out: list = []
    try:
        v = _get(doc, "GetFirstView")
    except Exception:                                                  # noqa: BLE001
        return out
    n = 0
    while v is not None and n < 500:                                   #: ★ 上限防死循环
        n += 1
        out.append(v)
        try:
            v = _get(v, "GetNextView")
        except Exception:                                              # noqa: BLE001
            break
    return out


def _real_views(doc) -> list:
    """★ **真视图**（排除"图纸本身"）—— 先向已真机验证过的 `DQ.list_views` 取名单，再按名对上对象。"""

    except Exception:                                                  # noqa: BLE001
        return []
    return [v for v in _chain(doc) if _vname(v) in names and _vname(v)]


# ── 判据：★ 全部是**可回读**的量，且每格调用前后各读一次 ────────────────────────
def _break_state(view) -> dict:
    """★ 视图的断裂状态读数。

    ★ 三个来源，口径各不相同，★ 所以**三个都读、都报**（让人看得见是哪一份变了）：
      | 键 | 官方成员 | 签名 | 口径 |
      |---|---|---|---|
      | `is_broken` | `IView.IsBroken()` | `System.bool IsBroken()` | **零参**可读：视图**是否已断开** |
      | `breaks_len` | `IView.GetBreakLines()` | `System.object GetBreakLines()` | **零参**可读：回 `Breaks` 集合 ⇒ **数组长度 = 断裂对数**（官方：一对断裂线算**一个** break）|
      | `count2` | `IView.GetBreakLineCount2(out Int32 Size)` | `System.int GetBreakLineCount2( out System.int Size )` | ★ **带 out** ⇒ 必须按 `sw_connect.out_i4()`（唯一真机验证通过的 byref 形态）传，★ 失败就如实记 `err` |
    """
    st: dict = {"is_broken": None, "breaks_len": None, "count2": None}
    try:
        st["is_broken"] = bool(_get(view, "IsBroken"))
    except Exception as exc:                                           # noqa: BLE001
        st["is_broken_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    try:
        arr = _get(view, "GetBreakLines")
        st["breaks_len"] = len(list(arr)) if arr is not None else 0
    except Exception as exc:                                           # noqa: BLE001
        st["breaks_len_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    try:
        st["count2"] = int(_get(view, "GetBreakLineCount2", C.out_i4(0)) or 0)
    except Exception as exc:                                           # noqa: BLE001
        #: ★ `out` 参数族在 pywin32 迟到绑定下常常到不了（与已知事实 47 同族）⇒ ★ 这里**预期会失败**，
        #: ★ 失败不是判负，只是"这条回读口径不可用"⇒ 判据退到上面两个零参成员。
        st["count2_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    return st


def _delta(before: dict, after: dict) -> dict:
    """★ 只对**两个零参口径**算增量（★ `count2` 可能一直是 None，不参与判定）。"""
    out = {}
    for k in ("breaks_len",):
        b, a = before.get(k), after.get(k)
        out[k + "_delta"] = (a - b) if isinstance(a, int) and isinstance(b, int) else None
    out["is_broken"] = [before.get("is_broken"), after.get("is_broken")]
    out["count2"] = [before.get("count2"), after.get("count2")]
    return out

def _build_part(sw, row: dict):
    """★ 建 80×60×10 mm 的板并**存盘**（★★ 存盘是硬要求，见 `PART_PATH` 那行注释）。"""
    doc = None
    try:
        doc = C.new_document(sw, "part")
        P.start_sketch(doc, plane_name="Front Plane")
        _get(_get(doc, "SketchManager"), "CreateCornerRectangle",
             -0.04, -0.03, 0.0, 0.04, 0.03, 0.0)
        P.end_sketch(doc)
        names = [n for n in (P.feature_names(doc) or [])
                 if n.startswith("草图") or n.lower().startswith("sketch")]
        if not names:
            row["extrude_err"] = "找不到草图特征名（feature_names 没回草图）"
            return None
        P.extrude_boss(doc, names[-1], 0.01)
        row["extrude_sketch"] = names[-1]
        row["saved"] = bool(C.save_document(doc, str(PART_PATH)))
        row["part_exists"] = PART_PATH.is_file()
        return doc if row.get("saved") else None
    except Exception as exc:                                           # noqa: BLE001
        row["part_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:180])
        return doc


def _drawing_with_view(sw, part_doc, row: dict, view_name="*Front"):
    """★ 建工程图 + 插一个模型视图 ⇒ `(drawing, 真视图, 视图名, 信息)`。

    ★ 前置（官方 Remarks 逐字）：`ModelName` 必须是**含 `.sldprt` 的完整路径**；
      `ViewName` 与模型视图名**完全一致**，标准视图名**带星号**；Loc* 单位是**米**。
    """
    info: dict = {}
    if part_doc is None:
        return None, None, None, {"err": "★ 前置未成立：零件文档不在（没建成/没存盘）"}
    try:

    except Exception as exc:                                           # noqa: BLE001

        info["call_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:180])
    views = _real_views(doc)
    info["view_count"] = len(views)
    return doc, (views[0] if views else None), (_vname(views[0]) if views else None), info


def _select_view_by_id(doc, view_name: str, row: dict):
    """★ 照官方例子取视图：`ActivateView(name)` → `SelectByID2(name,"DRAWINGVIEW",0,0,0,False,0,…)`
    → `SelectionManager.GetSelectedObject6(1, -1)`；★ 每一步都回读，**不回读到就如实说前置不成立**。"""
    got: dict = {"view_name": view_name}
    try:
        got["activate"] = bool(_get(doc, "ActivateView", str(view_name)))
    except Exception as exc:                                           # noqa: BLE001
        got["activate_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    _clear(doc)
    ext = _get(doc, "Extension")
    try:
        #: ★ 官方例子第 8 个实参写的是 `Nothing` ⇒ 空 Dispatch（已知事实 33：裸 None 会"类型不匹配"）。
        got["select_return_ignored"] = bool(
            ext.SelectByID2(str(view_name), "DRAWINGVIEW", 0.0, 0.0, 0.0,
                            False, 0, C.out_dispatch(None), 0))
    except Exception as exc:                                           # noqa: BLE001
        got["select_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    got["sel_by_mark"] = _sel_counts(doc)
    view = None
    try:
        view = _get(_get(doc, "SelectionManager"), "GetSelectedObject6", 1, -1)
    except Exception as exc:                                           # noqa: BLE001
        got["get_sel_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
    got["view_from_selection"] = view is not None
    got["view_name_readback"] = _vname(view) if view is not None else None
    #: ★★ **前置判据只看计数**（事实 8）⇒ 视图取不到 ⇒ 后面那一格**不作"能力不成"的证据**。
    got["preconditions_held"] = bool(view is not None
                                     and got.get("view_name_readback") == view_name)
    if view is None:
        row.setdefault("precondition_note", []).append(
            "★ 前置未成立：GetSelectedObject6 没取到视图（sel_by_mark=%s）" % got["sel_by_mark"])
    return view, got

def _selected(filters, label) -> bool:
    if not filters:
        return True
    return any(f.lower() in label.lower() for f in filters)


def main() -> int:
    """★ 跑格子（或筛选后的）⇒ 写 `output/break3-probe-evidence.json` 并打印判读。

    ⚠ ★★ **本文件是草稿、★ 本轮按指令没有执行过它** ⇒ 下面这些读数**一个都不存在**；
      ★ 跑法（★ 真机一律走弹窗护栏，★ 且**不由本文件的作者随手跑**）：
          python tools/sw_guard.py --timeout 1800 -- python <本文件改名为 .py 之后>
      ★ 只跑某几格：把标签子串当参数传，例如 `… python …py D A`
    """
    filters = [a for a in sys.argv[1:] if not a.startswith("-")]
    sw, _meta = C.connect_solidworks(wait_seconds=30)
    ev = {"at": time.strftime("%Y-%m-%dT%H:%M:%S"),
          "_": "IView.InsertBreak3（★ InsertBreak/2 已 Obsolete）—— ★ 每格先回读前置再调用；"
               "★ 判据 = GetBreakLines().len +1 ∧ 返回非 None ∧ BreakLine.Style/ShapeIntensity 可读",
          "spec": "specs/202-user-coverage-gaps",
          "official_signature": SIGNATURE_VERBATIM,
          "official_signature_arity": SIGNATURE_ARITY,
          "position_unit_assumption": POSITION_UNIT_ASSUMPTION,
          "filters": filters, "part_path": str(PART_PATH), "cases": []}
    try:
        #: ★ 铁律 2：未抑制的模态框会把会话挂死 ⇒ 跑之前先置位（★ 收尾必复位）
        try:
            sw.CommandInProgress = True
            ev["command_in_progress"] = True
        except Exception as exc:                                       # noqa: BLE001
            ev["command_in_progress"] = "%s: %s" % (type(exc).__name__, str(exc)[:120])
        for label, desc, fn in CASES:
            if not _selected(filters, label):
                continue
            t0 = time.time()
            try:
                case = fn(sw)
            except Exception as exc:                                   # noqa: BLE001
                #: ★ 不许吞异常：★ 异常本身进 dict
                case = {"tag": label, "ok": False,
                        "err": "%s: %s" % (type(exc).__name__, str(exc)[:240])}
            case["label"] = label
            case["desc"] = desc
            case["seconds"] = round(time.time() - t0, 2)
            ev["cases"].append(case)
            print("  %-3s %-40s ok=%-6s pre=%-6s %ss"
                  % (label, desc, case.get("ok"), case.get("preconditions_held"),
                     case.get("seconds")))
            for k in ("delta", "break_before", "break_after", "args", "breakline_props",
                      "callable_counters", "enum_missing", "err", "call_err", "note"):

    finally:
        try:
            sw.CommandInProgress = False
        except Exception:                                              # noqa: BLE001
            pass
    ev["passed"] = sum(1 for c in ev["cases"] if c.get("ok"))
    ev["total"] = len(ev["cases"])
    ev["preconditions_held_count"] = sum(1 for c in ev["cases"] if c.get("preconditions_held"))
    OUT_JSON.parent.mkdir(parents=True, exist_ok=True)
    OUT_JSON.write_text(json.dumps(ev, ensure_ascii=False, indent=1, default=str) + "\n",
                        encoding="utf-8")
    print("  判读：成 %d / %d（★ 前置成立的 %d 格）"
          % (ev["passed"], ev["total"], ev["preconditions_held_count"]))
    print("  ⚠ ★ `ok=False` 且 `preconditions_held=False` 的格子**不能**当「能力不成」的证据。")
    print("  产物：%s" % OUT_JSON)
    return 0
