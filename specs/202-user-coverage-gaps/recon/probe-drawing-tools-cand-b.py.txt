def _vname(view) -> str:
    """★ 视图名（`GetName2` → 退回 `Name`，与 `sw_drawing_query` 同口径）。"""
    for attr in ("GetName2", "Name"):
        try:
            value = _get(view, attr)
            if value:
                return str(value)
        except Exception:                                             # noqa: BLE001
            continue
    return ""


def _view_chain(doc):
    """★ 视图链：`GetFirstView()` → `GetNextView()`（★ 含"图纸本身"那个伪视图）。"""
    out = []
    try:
        view = _get(doc, "GetFirstView")
    except Exception as exc:                                          # noqa: BLE001
        return out, "%s: %s" % (type(exc).__name__, str(exc)[:120])
    guard = 0
    while view is not None and guard < 500:                           #: ★ 防死循环
        guard += 1
        out.append(view)
        try:
            view = _get(view, "GetNextView")
        except Exception:                                             # noqa: BLE001
            break
    return out, None


def _real_views(doc):
    """★ **真视图**（★ 排除"图纸本身"那个伪视图）⇒ `(views, names, err)`。"""
    try:
        listed = [str((r or {}).get("name") or "") for r in (DQ.list_views(doc) or [])]
    except Exception as exc:                                          # noqa: BLE001
        return [], [], "取视图清单失败：%s: %s" % (type(exc).__name__, str(exc)[:120])
    chain, err = _view_chain(doc)
    if err:
        return [], [], err
    return [v for v in chain if _vname(v) in listed], listed, None

def _counts_totals(doc) -> dict:
    """★ 注解读数：真视图与整条链各一份（★ 有些注解挂在图纸上而不挂在视图里 ⇒ 两份都报）。"""
    real, _names, err = _real_views(doc)
    chain, _e = _view_chain(doc)
    state: dict = {"err": err}
    for label, views in (("real", real), ("all", chain)):
        totals = {}
        for key, member in _COUNTERS:
            vals = []
            for view in views:
                try:
                    vals.append(int(_get(view, member) or 0))
                except Exception:                                     # noqa: BLE001
                    pass
            totals[key] = sum(vals) if vals else None
        totals["count"] = len(views)
        state[label] = totals
    state["per_view"] = [{"name": _vname(v),
                          **{k: _safe_count(v, m) for k, m in _COUNTERS}}
                         for v in real]
    return state


def _safe_count(view, member):

def _delta(before, after, key, bucket="all"):
    """★ 增量（★ 任一份量不到就回 None ⇒ 判据降级，不猜 0）。"""
    try:
        b = before[bucket][key]
        a = after[bucket][key]
    except Exception:                                                 # noqa: BLE001
        return None
    if b is None or a is None:
        return None
    return int(a) - int(b)

def _enum(enum_name: str, member: str):
    """★ **在官方枚举页里现场取枚举值** ⇒ `(value|None, 来源或原因)`。

    ★★ 沿用 `tools/prove_drawing_batch.py` 的 `_enum`（同一份解析规则）：只认页里
    `| Member | Value |` 那张表；Value 列形如 `12` / `1 = Upper-left corner` / `4 or 0x4; 说明…`
    ⇒ 先试 `0x…` 再试十进制。★ 读不到 ⇒ `(None, 原因)` ⇒ **调用方据此判前置未成立，不猜**。
    """
    key = (enum_name, member)
    if key in _ENUM_CACHE:
        return _ENUM_CACHE[key]
    page = SWAPI_ENUMS / ("%s.md" % enum_name)
    value: int | None = None
    note: str | None = None
    if not page.is_file():
        note = "★ 官方枚举页不在本机：%s" % page
    else:
        try:
            text = page.read_text(encoding="utf-8", errors="replace")
        except OSError as exc:
            text, note = "", "★ 读官方枚举页失败：%s" % str(exc)[:120]
        if text:
            for line in text.splitlines():
                if not line.startswith("|"):
                    continue
                cells = [c.strip() for c in line.strip().strip("|").split("|")]
                if len(cells) < 2 or cells[0] != member:
                    continue
                raw = cells[1]
                match = re.match(r"^0x([0-9a-fA-F]+)", raw) or re.match(r"^(-?\d+)", raw)
                if match:
                    value = int(match.group(1), 16) if raw.lower().startswith("0x") \
                        else int(match.group(1))
                    note = str(page)
                else:
                    note = "★ %s 的 Value 列不是数字（%r）⇒ 不解析、不猜" % (member, raw[:60])
                break
            else:
                note = "★ 官方枚举页 %s 里没有成员 %s ⇒ 不猜" % (page.name, member)
    _ENUM_CACHE[key] = (value, note)
    return value, note


def _enum_or_note(row: dict, enum_name: str, member: str, label: str = ""):
    """★ 取枚举值；取不到就**记进 row** 并回 None（★ 调用方据此跳过该格，不硬凑）。"""
    value, note = _enum(enum_name, member)
    if value is None:
        row.setdefault("enum_missing", []).append(
            "%s.%s%s ⇒ %s" % (enum_name, member, ("（%s）" % label) if label else "", note or ""))
    else:
        row.setdefault("enum_read", {})["%s.%s" % (enum_name, member)] = value
    return value


def _find_files(patterns, roots=TEMPLATE_ROOTS):
    """★ 在若干根目录下按文件名找（★ 用到才找，找不到就**如实回空**）。"""
    hits: list[str] = []
    for root in roots:
        base = Path(root)
        if not base.is_dir():
            continue
        for pattern in patterns:
            try:
                hits.extend(sorted(str(p) for p in base.rglob(pattern)))
            except OSError:
                continue
    return hits


# ── spec 装载与契约回读（★ 任务硬要求①：按 ToolSpec 的契约调） ───────────────
def load_batch75_specs():
    """★ 取 batch75 的 8 条 spec。

    ★ 主路：`register_capabilities.all_specs()`（★ 任务指定的那条），按 `TOOL_NAMES` 过滤；
    ★ 备路：直接 `tool_specs_batch75.SPECS_BATCH75`（★ 只在主路抛错时用，并把用了哪条写进证据）。
    ★ **两条都读不到** ⇒ 返回 `(specs, info)` 且 `info["ok"] = False` ⇒ 整份探针判"前置未成立"。
    """
    info: dict = {"source": None, "missing_tools": list(TOOL_NAMES), "error": None}
    try:
        specs = [s for s in RC.all_specs() if getattr(s, "name", "") in TOOL_NAMES]
        if specs:
            info["source"] = "register_capabilities.all_specs()"
            info["ok"] = True
            info["missing_tools"] = [n for n in TOOL_NAMES
                                     if n not in {s.name for s in specs}]
            return {s.name: s for s in specs}, info
        info["error"] = "★ all_specs() 里没有这 8 条（缺：%s）" % info["missing_tools"]
    except Exception as exc:                                          # noqa: BLE001
        info["error"] = "all_specs() 抛错：%s: %s" % (type(exc).__name__, str(exc)[:200])
    try:
        import tool_specs_batch75 as B75                                # noqa: PLC0415
        specs = [s for s in (B75.SPECS_BATCH75 or []) if getattr(s, "name", "") in TOOL_NAMES]
        info["source"] = "tool_specs_batch75.SPECS_BATCH75（备路）"
        info["ok"] = bool(specs)
        info["missing_tools"] = [n for n in TOOL_NAMES if n not in {s.name for s in specs}]
        return {s.name: s for s in specs}, info
    except Exception as exc:                                          # noqa: BLE001
        info["error"] = "%s ｜ 备路也抛错：%s: %s" % (
            info.get("error") or "", type(exc).__name__, str(exc)[:160])
        info["ok"] = False
        return {}, info


def contract_of(spec) -> dict:
    """★ 把一条 spec 的**契约**回读成可核对的 dict（★ 不改一个字）。"""
    return {
        "tool_name": spec.name,
        "title": spec.title,
        "module": spec.module,
        "function": spec.function,
        "inject": effective_inject(spec),
        "params": [{"name": p.name, "kind": getattr(p.kind, "__name__", str(p.kind)),
                    "required": bool(p.is_required), "description": (p.description or "")[:120]}
                   for p in spec.params],
        "param_alias": dict(getattr(spec, "param_alias", {}) or {}),
        "registry_only_hooks": [name for name in ("extra_op", "derived_args", "pre_call")
                                if getattr(spec, name, None) is not None],
        "omitted_params": list(getattr(spec, "omitted_params", ()) or ()),
    }


def _build_kwargs(spec, args: dict) -> dict:
    """★ 按 `spec.params` 逐个取实参（★ 规则照抄 `tool_registry.py:492-498`）。

    ★ `param_alias[名] is None` ⇒ 该参数由解析器消费、**不进**目标函数；
    ★ 未列出的参数**同名直传**；★ 必填参数没准备 ⇒ **抛**（不静默补默认）。
    """
    kwargs: dict = {}
    missing: list[str] = []
    alias = dict(getattr(spec, "param_alias", {}) or {})
    for p in spec.params:
        target = alias.get(p.name, p.name) if alias else p.name
        if target is None:
            continue
        if p.name not in args:
            missing.append(p.name)
            continue
        kwargs[target] = args[p.name]
    if missing:
        raise RuntimeError("工具 %s 的参数没准备：%s（★ 按 spec.params 逐条核对，不猜）"
                           % (spec.name, missing))
    return kwargs


def call_tool(spec, args: dict):
    """★ **经工具层**调一条工具 ⇒ `(结果, 调用信息)`。

    ★ 四步与 `make_tool_function` 的 `op()` 同序：
      ① 组装 kwargs（见 `_build_kwargs`）② 注入宿主对象 ③ `__import__("scripts.%s")`
      ④ `getattr(mod, spec.function)(**kwargs)`
    ★ `response_format` **不传**（注册器也不传 —— 它只遍历 `spec.params`）。
    """
    info: dict = {"tool_name": spec.name, "module": spec.module,
                  "function": spec.function, "args_given": dict(args)}
    kwargs = _build_kwargs(spec, args)
    inject_key = effective_inject(spec)
    if inject_key:
        resolver = (RC.RESOLVERS or {}).get(inject_key)
        if resolver is None:
            raise RuntimeError(
                "工具 %s 声明了 inject=%r，但注册时没给对应解析器（已给：%r）"
                % (spec.name, inject_key, sorted((RC.RESOLVERS or {}).keys())))
        shim = types.SimpleNamespace(**kwargs)      # ★ 与注册器一样，把 params 传给解析器
        host = resolver(shim)
        kwargs[inject_key] = host                   # ★ 注入键 = inject 的值（`tool_registry.py:507`）
        info["inject_key"] = inject_key
        info["inject_host_title"] = _safe_str(host, "GetTitle")
        info["inject_host_path"] = _safe_str(host, "GetPathName")
        info["inject_host_type"] = _safe_count(host, "GetType")
    info["kwargs_keys"] = sorted(kwargs)
    mod = __import__("scripts.%s" % spec.module, fromlist=[spec.function])
    fn = getattr(mod, spec.function)
    info["resolved_callable"] = "%s.%s" % (mod.__name__, spec.function)
    return fn(**kwargs), info


def _safe_str(obj, member):
    try:
        return str(_get(obj, member) or "")
    except Exception:                                                 # noqa: BLE001
        return None


def check_inject_is_active_doc(sw, host, info: dict) -> None:
    """★ 回读「注入进去的就是**当前活动**工程图吗」—— ★ 这是 `inject="drawing_model"`
    的前提（`RESOLVERS["drawing_model"] = resolve_model` = **活动文档**）。"""
    try:
        active = sw.ActiveDoc
    except Exception as exc:                                          # noqa: BLE001
        info["inject_active_check"] = "读 ActiveDoc 失败：%s: %s" % (type(exc).__name__,

    info["inject_active_title"] = _safe_str(active, "GetTitle")
    info["inject_active_path"] = _safe_str(active, "GetPathName")
    info["inject_matches_active_doc"] = bool(
        info.get("inject_host_title") and info.get("inject_host_title") == info["inject_active_title"])

def build_saved_part(sw, row: dict):
    """★ 建一个**带孔**的板并**存盘**（★ 存盘是硬前置 —— 见下）。

    ★★ 为什么必须存盘：官方 `CreateDrawViewFromModelView3` 的 Remarks 逐字要求 `ModelName` 是
      **含 `.sldprt`/`.sldasm` 扩展名的完整路径** ⇒ `create_projected_view` 的 `preconditions_held`
      就是按这个判的（`scripts/sw_drawing_expose.py:73`）。
    ★ 板上打一个孔：让"孔表 / 中心符号线"那两格有圆边可用。
    ★ 结构逐字沿用 `tools/prove_drawing_batch.py::_build_part`（★ 已真机跑过）。
    """
    doc = C.new_document(sw, "part")
    try:
        P.start_sketch(doc, plane_name="Front Plane")
        _get(_get(doc, "SketchManager"), "CreateCornerRectangle",
             -0.04, -0.03, 0.0, 0.04, 0.03, 0.0)
        P.end_sketch(doc)
        names = [n for n in (P.feature_names(doc) or [])
                 if n.startswith("\u8349\u56fe") or n.lower().startswith("sketch")]
        if not names:
            row["part_err"] = "找不到草图特征名（feature_names 没回草图）"
            return None
        P.extrude_boss(doc, names[-1], 0.01)
        row["extrude_sketch"] = names[-1]
    except Exception as exc:                                          # noqa: BLE001
        row["part_err"] = "建板失败：%s: %s" % (type(exc).__name__, str(exc)[:180])
        return None
    try:
        rep = HF.create_hole_wizard_at_selection_report(doc, "+z", "M8")
        row["hole"] = {k: rep.get(k) for k in ("ok", "selection_mark") if k in rep}
    except Exception as exc:                                          # noqa: BLE001
        #: ★ 孔不成**不阻断**：只是"孔表 / 中心符号线"的前置更弱 ⇒ 如实记下来。
        row["hole_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:180])
    try:
        row["part_saved"] = bool(C.save_document(doc, str(PART_PATH)))
    except Exception as exc:                                          # noqa: BLE001
        row["part_save_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:180])
        row["part_saved"] = False
    return doc if row.get("part_saved") else None


def find_drawing_template() -> str | None:
    """★ 工程图模板（★ 现查；查不到就如实回 None ⇒ 那格判前置未成立）。"""
    for candidate in DRAWING_TEMPLATES:

    hits = _find_files(("gb_a3.drwdot", "*.drwdot"), (r"C:\ProgramData\SOLIDWORKS",))

def new_activated_drawing(sw, row: dict):
    """★ 新建工程图 → **激活** → 回读"活动文档就是它" ⇒ `(drawing|None, info)`。

    ★ 为什么必须激活：`RESOLVERS["drawing_model"] = resolve_model` 取的是 **ActiveDoc**
      ⇒ ★ 不激活的话，工具层会拿到**上一张文档**（或报错），那就不是"经工具层"了。
    ★ 激活用官方 `ISldWorks::ActivateDoc3(Name, UseUserPreferences, Option, out Errors)`
      （`Errors` 用 `C.out_i4()` 包 —— 官方那四个 out 参数形态只有这一种真机验证过）。
    """
    template = find_drawing_template()
    row["drawing_template"] = template
    if not template:
        row["drawing_template_found"] = False
        return None, {"err": "★ 前置未成立：找不到 `.drwdot` 工程图模板"}
    row["drawing_template_found"] = True
    try:
        doc = C.new_document(sw, "drawing", template_path=str(template))
    except Exception as exc:                                          # noqa: BLE001
        return None, {"err": "新建工程图失败：%s: %s" % (type(exc).__name__, str(exc)[:180])}
    info: dict = {"template": str(template)}
    errors = C.out_i4(0)
    try:
        _get(sw, "ActivateDoc3", str(_get(doc, "GetTitle") or ""), False, 0, errors)
        info["activate_doc3_returned"] = True
    except Exception as exc:                                          # noqa: BLE001
        info["activate_doc3_error"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    try:
        info["activate_errors_code"] = int(getattr(errors, "value", -1))
    except Exception:                                                 # noqa: BLE001
        info["activate_errors_code"] = None
    info["active_title"] = _safe_str(sw.ActiveDoc, "GetTitle")
    info["is_active"] = bool(info["active_title"]) and \
        info["active_title"] == _safe_str(doc, "GetTitle")
    info["doc_type"] = _safe_count(doc, "GetType")           #: ★ 3 = `swDocDRAWING`
    try:
        info["sheet_name"] = DQ.current_sheet_name(doc)
        info["views_before"] = _view_count(doc)
    except Exception as exc:                                          # noqa: BLE001
        info["sheet_err"] = "%s: %s" % (type(exc).__name__, str(exc)[:160])
    if not info.get("is_active"):
        info["err"] = "★ 激活没生效（ActiveDoc=%r，期望=%r）⇒ 这一格不算工具层路径" % (
            info["active_title"], _safe_str(doc, "GetTitle"))
    return doc, info


def setup_views_through_tool(sw, specs, doc, row: dict, projection="third_angle"):
    """★ 用**工具层**的那条投影视图把视图建起来（★ 后面几格的前置）。

    ★ 为什么经工具而不是直接 `Create3rdAngleViews2`：① 后面几格的前置**就是**这条工具的产物；
      ② ★ 这样"工具层能不能建视图"与"别的工具能不能在有视图的图上跑"是**同一个**读数。
    ★ 用 `third_angle`（官方成员 `Create3rdAngleViews2` 在本机官方镜像里**有**），
      ★ `first_angle` 单独一格（见 T2）⇒ 前置不因那条成员名问题而垮掉。
    """
    spec = specs.get("solidworks_drawing_create_projected_view")
    if spec is None:
        row["setup"] = {"err": "spec 里没有投影视图那条"}
        return False
    before = _view_count(doc)
    try:
        result, info = call_tool(spec, {"part_path": str(PART_PATH), "projection": projection})
    except Exception as exc:                                          # noqa: BLE001
        row["setup"] = {"err": "%s: %s" % (type(exc).__name__, str(exc)[:200])}
        return False
    after = _view_count(doc)
    views, names, err = _real_views(doc)
    row["setup"] = {"projection": projection, "call": info, "raw": result,
                    "views_before": before, "views_after": after,
                    "view_names": names, "view_chain_err": err}
    row["setup"]["ok"] = bool(before is not None and after is not None and after > before)
    return bool(row["setup"]["ok"])

def case_t3_section_view(sw, specs):

    而能力层**不代选**、只回读选区计数并把 `preconditions_held` 带回（★ 事实 496）。

    ★ 本格照官方例子 `Create_Section_View_at_Specified_Location_Example_VB` 的**调用序**做前置：
      `ActivateView` → `SelectByID2(视图名,"DRAWINGVIEW")` → `ClearSelection2` →

    ★ 判据：**视图数 +1**（★ 且前置成立才算）。"""

           "ok": False, "verdict": PRECONDITION_NOT_MET}

    try:

        if part is None or not PART_PATH.is_file():

        _clear(doc)

    except Exception as exc:                                          # noqa: BLE001
        row["err"] = "%s: %s" % (type(exc).__name__, str(exc)[:240])
        row["verdict"] = TRUE_NEGATIVE
    finally:

def main() -> int:
    """★ 跑（或筛选后跑）全部格子 ⇒ 写 `output/drawing-tools-e2e-evidence.json`。"""
    filters = [a for a in sys.argv[1:] if not a.startswith("-")]
    evidence: dict = {
        "at": time.strftime("%Y-%m-%dT%H:%M:%S"),
        "_": "★ 工程图 8 条工具的**工具层**端到端探针 —— ★ 每格按 ToolSpec 契约组装 kwargs；"
             "★ 三态判词：built（建成）/ precondition_not_met（前置未成立，写明缺什么）/"
             "true_negative（真阴性）；★ ★ ok=False 且 verdict=precondition_not_met 的格子"
             "**不能**当「能力不成」的证据",
        "spec": "specs/202-user-coverage-gaps",
        "how_called": "★ 手工按 spec.params 组装 kwargs + 照 tool_registry.make_tool_function 的"
                      "最后两行 __import__(\"scripts.%s\" % spec.module) + getattr(...)(**kwargs)；"
                      "★ 注入走 register_capabilities.RESOLVERS[spec.inject]（与真机同源）",
        "run_with_guard": "python tools/sw_guard.py --timeout 1800 -- python <本脚本>",
        "filters": filters,
        "temp_files": {"part": str(PART_PATH), "drawing": str(DRAW_PATH)},
        "cases": [],
    }

    specs, spec_info = load_batch75_specs()
    evidence["spec_loading"] = spec_info
    evidence["contract"] = {
        "tool_count": len(specs),
        "missing_tools": spec_info.get("missing_tools"),
        "tools": [contract_of(specs[name]) for name in TOOL_NAMES if name in specs],
    }
    #: ★ 自认的保真度边界：注册器那三个钩子本稿**没有**实现 ⇒ 用了就不算工具层等价路径。
    evidence["contract"]["registry_only_hooks_present"] = sorted(
        {hook for c in evidence["contract"]["tools"] for hook in c["registry_only_hooks"]})
    if not specs:
        evidence["summary"] = {BUILT: 0, PRECONDITION_NOT_MET: len(CASES), TRUE_NEGATIVE: 0,
                                "total": 0,
                                "note": "★ spec 没装载上 ⇒ 全部格子判「前置未成立」"
                                        "（★ 这是探针自身的前置问题，**不是**能力问题）"}
        _write(evidence)
        print("  ★ spec 没装载上：%s" % spec_info.get("error"))
        print("  产物：%s" % OUT_JSON)
        return 0

    sw, _model = C.connect_solidworks(wait_seconds=30)
    try:
        #: ★ 铁律 2：未抑制的模态框会把会话挂死 ⇒ 跑之前置位（★ 收尾必复位）。
        try:
            sw.CommandInProgress = True
            evidence["command_in_progress"] = True
        except Exception as exc:                                      # noqa: BLE001
            evidence["command_in_progress"] = "%s: %s" % (type(exc).__name__, str(exc)[:140])
        for label, desc, fn in CASES:
            if not _selected(filters, label):
                continue
            started = time.time()
            try:
                case = fn(sw, specs)
            except Exception as exc:                                  # noqa: BLE001
                #: ★ 不许吞异常：★ 异常本身进 dict（★ 任务硬要求）。
                case = {"tag": label, "ok": False, "verdict": TRUE_NEGATIVE,
                        "err": "%s: %s" % (type(exc).__name__, str(exc)[:240])}
            case["label"] = label
            case["desc"] = desc
            case["seconds"] = round(time.time() - started, 2)
            evidence["cases"].append(case)
            print("  %-4s %-40s %-22s ok=%-6s %ss"
                  % (label, desc, case.get("verdict"), case.get("ok"), case.get("seconds")))
            for key in ("view_delta", "note_delta_all", "table_delta_all", "center_delta_all",
                        "readback_changed", "capability_status", "capability_error_code",
                        "preconditions_held", "tool_ok", "tool_preconditions_held",
                        "enum_missing", "note", "err"):
                if key in case:
                    print("        %-26s %s" % (key, str(case[key])[:170]))
    finally:
        try:
            sw.CommandInProgress = False
        except Exception:                                             # noqa: BLE001
            pass

    cases = evidence["cases"]
    evidence["summary"] = {
        BUILT: sum(1 for c in cases if c.get("verdict") == BUILT),
        PRECONDITION_NOT_MET: sum(1 for c in cases if c.get("verdict") == PRECONDITION_NOT_MET),
        TRUE_NEGATIVE: sum(1 for c in cases if c.get("verdict") == TRUE_NEGATIVE),
        "total": len(cases),
    }
    #: ★ **工具自己回的 ok 与探针读数不一致**的那些格子单列 —— ★ 这是工具层最该看的一栏
    #:   （例如 `create_projected_view`：能力层回的是 `status/created` 而不是 `ok`，
    #:   而包装层 `_result` 取 `payload["ok"]` ⇒ **建成时工具的 ok 也会是 False**）。
    evidence["tool_ok_vs_readback"] = [
        {"tool": c.get("label"), "tool_ok": c.get("tool_ok"), "ok": c.get("ok"),
         "note": "★ 工具的 ok 与读数判定不一致 —— 按读数解释，★ 如实留档"}
        for c in cases
        if isinstance(c.get("tool_ok"), bool) and bool(c.get("tool_ok")) != bool(c.get("ok"))
    ]
    _write(evidence)
    print("  判读：建成 %d · 前置未成立 %d · 真阴性 %d（共 %d 格）"
          % (evidence["summary"][BUILT], evidence["summary"][PRECONDITION_NOT_MET],
             evidence["summary"][TRUE_NEGATIVE], evidence["summary"]["total"]))
    print("  ⚠ ★ verdict=precondition_not_met 的格子**不能**当「能力不成」的证据。")
    print("  ⚠ ★ verdict=true_negative 的格子请连 `note` 一起读（★ 分清是能力层还是工具层）。")
    print("  产物：%s" % OUT_JSON)
    return 0


def _write(evidence: dict) -> None:
    OUT_JSON.parent.mkdir(parents=True, exist_ok=True)
    OUT_JSON.write_text(json.dumps(evidence, ensure_ascii=False, indent=1, default=str) + "\n",
                        encoding="utf-8")


if __name__ == "__main__":
    raise SystemExit(main())
